```xml
<?xml version="1.0"?>
<!DOCTYPE map [
  <!ENTITY % pair "(key, value)">
  <!ELEMENT map (pairset, comment?)>
  <!ELEMENT pairset (pair+)>
  <!ELEMENT pair %pair;>
  <!ELEMENT key (#PCDATA)>
  <!ELEMENT value (#PCDATA|list)*>
  <!ELEMENT list (item*)>
  <!ELEMENT item (#PCDATA)>
  <!ELEMENT comment (#PCDATA)>
  <!ATTLIST pair kind (scalar|compound) "scalar">
  <!ATTLIST item n NMTOKEN #IMPLIED>
]>
<map>
  <pairset>
    <pair kind="scalar">
      <key>path</key>
      <value>/var/tmp</value>
    </pair>
    <pair kind="compound">
      <key>options</key>
      <value>
        <list>
          <item n="0">fast</item>
          <item n="1">safe</item>
          <item n="2"/>
        </list>
      </value>
    </pair>
  </pairset>
  <comment>after the final pair</comment>
</map>
```

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE report [
  <!ELEMENT report (section|separator)+>
  <!ELEMENT section (heading, data)>
  <!ELEMENT heading (#PCDATA)>
  <!ELEMENT data (#PCDATA|line|ref)*>
  <!ELEMENT line (#PCDATA)>
  <!ELEMENT ref EMPTY>
  <!ELEMENT separator EMPTY>
  <!ATTLIST section number ID #REQUIRED>
  <!ATTLIST ref target IDREF #REQUIRED>
  <!ATTLIST separator style (thin|thick) #IMPLIED>
  <!ENTITY intro "alpha &amp; omega">
  <!ENTITY quoted "&quot;quoted&quot;">
]>
<report>
  <section number="sA">
    <heading>&intro;</heading>
    <data>
      first<line>second&#x0A;third</line>
      <ref target="sA"/>end &quoted;
    </data>
  </section>
  <separator style="thick"/>
  <section number="sB">
    <heading>Δ</heading>
    <data><![CDATA[less-than: <, greater-than: >, ampersand: &]]></data>
  </section>
</report>
```
