]>

    <field name="last_seen"/>

  </from>
  <where>
    <and>
      <condition field="status" operator="=" value="active"/>
      <or>
        <condition field="temperature" operator="&gt;" value="80"/>

          <condition field="region" operator="=" value="test"/>
        </not>
      </or>
