  <!ELEMENT archive (file*)>
  <!ELEMENT file (#PCDATA)>

            name CDATA #REQUIRED
            encoding NOTATION #IMPLIED

]>
<?archive mode="index"?>

  <file name="empty" encoding="bin"/>
  <file name="linked" payload="sample">metadata</file>
