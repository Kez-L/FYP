<?xml version="1.0"?>
<!DOCTYPE config [
  <!ELEMENT config (module+, checksum?)>
  <!ELEMENT module (name, option*)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT option EMPTY>
  <!ELEMENT checksum (#PCDATA)>
  <!ATTLIST module
    uid ID #REQUIRED
    depends IDREFS #IMPLIED
    enabled (yes|no) "yes">
  <!ATTLIST option
    key CDATA #REQUIRED
    type (string|number|boolean) #IMPLIED
    value CDATA #REQUIRED>
  <!ATTLIST checksum
    algorithm NOTATION (sha1|md5) #IMPLIED>
  <!NOTATION sha1 SYSTEM "urn:hash:sha1">
  <!NOTATION md5 SYSTEM "urn:hash:md5">
  <!ENTITY nbsp "&#160;">
  <!ENTITY hex "&#x41;&#x42;&#x43;">
]>
<config>
  <module uid="core" enabled="yes">
    <name>engine</name>
    <option key="path" type="string" value="/tmp/&nbsp;cache"/>
    <option key="threads" type="number" value="08"/>
  </module>
  <module uid="ui" depends="core missing" enabled="no">
    <name>&hex;</name>
    <option key="visible" type="boolean" value="false"/>
  </module>
  <checksum algorithm="md5">9f86d081884c7d659a2feaa0c55ad015</checksum>
