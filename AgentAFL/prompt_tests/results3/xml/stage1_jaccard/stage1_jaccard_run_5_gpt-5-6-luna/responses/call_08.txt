<?xml version="1.0"?>
<!DOCTYPE exchange [
  <!ELEMENT exchange (request, response)>
  <!ELEMENT request (header*, payload)>
  <!ELEMENT response (header*, payload|fault)>
  <!ELEMENT header EMPTY>
  <!ELEMENT payload ANY>
  <!ELEMENT fault (code, reason, detail?)>
  <!ELEMENT code (#PCDATA)>
  <!ELEMENT reason (#PCDATA)>
  <!ELEMENT detail (#PCDATA)>
  <!ATTLIST header name CDATA #REQUIRED value CDATA #REQUIRED>
  <!ATTLIST payload format (xml|text|binary) #IMPLIED>
  <!ENTITY quote "&quot;">
  <!ENTITY apos "&apos;">
  <!ENTITY nested "A&amp;B&lt;C&gt;">
]>

  <request>
    <header name="Accept" value="application/xml"/>
    <header name="X-Flags" value="one&#x20;two&#x09;three"/>

  </request>
  <response>
    <header name="Content-Type" value="text/plain"/>
    <fault>
      <code>422</code>
