    <!ELEMENT option (#PCDATA)>

    <!ELEMENT feature (switch|path|pattern)*>
    <!ELEMENT switch EMPTY>
    <!ELEMENT path (#PCDATA)>
    <!ELEMENT pattern (#PCDATA)>

]>
