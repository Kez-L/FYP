```xml
<!DOCTYPE catalog [
  <!ELEMENT catalog (entry | group)+>
  <!ELEMENT group (entry*)>
  <!ELEMENT entry (#PCDATA | link)*>
  <!ELEMENT link EMPTY>
  <!ATTLIST entry id ID #IMPLIED status CDATA "new">
  <!ATTLIST link href CDATA #REQUIRED>
  <!NOTATION binary SYSTEM "application/octet-stream">
  <!ENTITY payload SYSTEM "data.bin" NDATA binary>
]>
<catalog>
  <group>
    <entry id="e1">Alpha<link href="urn:test:alpha"/>Omega</entry>
    <entry status="archived">Before &amp; after</entry>
  </group>
  <entry id="e2">&payload;</entry>
</catalog>
```

```xml
<!DOCTYPE envelope [
  <!ELEMENT envelope ANY>
  <!ENTITY % declarations '
    <!ELEMENT packet (header?, body, checksum?)>
    <!ELEMENT header (#PCDATA)>
    <!ELEMENT body (chunk*)>
    <!ELEMENT chunk (#PCDATA)>
    <!ELEMENT checksum EMPTY>
    <!ATTLIST checksum algorithm NMTOKEN #IMPLIED>
  '>
  %declarations;
  <!ENTITY % conditional "INCLUDE">
  <![%conditional;[
    <!ENTITY marker "inside-section">
  ]]>
]>
<envelope>
  <packet>
    <header>&marker;</header>
    <body>
      <chunk>first</chunk>
      <chunk>second <![CDATA[ raw <data> ]]></chunk>
      <chunk>third &quot;value&quot;</chunk>
    </body>
    <checksum algorithm="sha256"/>
  </packet>
</envelope>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE report [
  <!ELEMENT report (section)>
  <!ELEMENT section (title, paragraph*, table?)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT paragraph (#PCDATA | emphasis | break)*>
  <!ELEMENT emphasis (#PCDATA)>
  <!ELEMENT break EMPTY>
  <!ELEMENT table (tr+)>
  <!ELEMENT tr (td+)>
  <!ELEMENT td (#PCDATA | emphasis)*>
  <!ATTLIST section mode (draft|final) "draft">
  <!ENTITY nbsp "&#160;">
]>
<report>
  <section mode="final">
    <title>Quarterly &nbsp; Report</title>
    <paragraph>Some <emphasis>important</emphasis> text<break/>continues.</paragraph>

      <tr><td>1</td><td><emphasis>A</emphasis>B</td></tr>
```
