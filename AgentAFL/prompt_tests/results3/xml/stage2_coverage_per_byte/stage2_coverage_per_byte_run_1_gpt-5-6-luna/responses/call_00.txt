```xml
    <name>alpha</name>
    <price>12.50</price>
```

```xml
<!DOCTYPE report [
  <!ELEMENT report (section)>
  <!ELEMENT section (title, entry*)>
  <!ATTLIST section xml:space (default|preserve) "default">
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT entry (key, value)>
  <!ELEMENT key (#PCDATA)>
  <!ELEMENT value (#PCDATA|br)*>
  <!ELEMENT br EMPTY>
]>
<report>
  <section xml:space="preserve">
    <title>Summary</title>
    <entry><key>A</key><value>left<br/>right</value></entry>
    <entry><key>B</key><value><![CDATA[&broken;]]></value></entry>
  </section>
</report>
```
