<?xml version="1.0"?>
<!DOCTYPE catalog [
  <!NOTATION svg SYSTEM "image/svg+xml">
  <!ENTITY logo SYSTEM "logo.svg" NDATA svg>
  <!ENTITY nbsp "&#160;">
  <!ELEMENT catalog (entry+, footer?)>
  <!ELEMENT entry (title, summary?, link*)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT summary (#PCDATA|em|code)*>
  <!ELEMENT em (#PCDATA)>
  <!ELEMENT code (#PCDATA)>
  <!ELEMENT link EMPTY>
  <!ELEMENT footer (#PCDATA)>
  <!ATTLIST catalog version NMTOKEN #REQUIRED>
  <!ATTLIST entry id ID #REQUIRED status (new|old|archived) "new">
  <!ATTLIST link href URI #REQUIRED rel CDATA #IMPLIED>
]>
<catalog version="2.0">
  <entry id="e1">
    <title>Symbols &amp; &nbsp; &lt;test&gt;</title>
    <summary>Use <em>carefully</em> with <code>CDATA</code>.</summary>
    <link href="urn:item:one" rel="alternate"/>
  </entry>
  <entry id="e2" status="archived">
    <title>Second entry</title>
    <link href="#e1"/>
  </entry>
  <footer>generated &logo;</footer>
</catalog>

  <!ELEMENT cell (#PCDATA)>

]>

  <row index="0" checksum="a&amp;b">
    <cell column="0" type="text">north</cell>
