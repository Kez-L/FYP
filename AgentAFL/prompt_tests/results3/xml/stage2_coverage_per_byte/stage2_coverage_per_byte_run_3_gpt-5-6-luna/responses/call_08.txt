<?xml version="1.1"?>
<!DOCTYPE response [
  <!ELEMENT response (status, data)>
  <!ELEMENT status (#PCDATA)>
  <!ELEMENT data (row*)>
  <!ELEMENT row (cell*)>
  <!ELEMENT cell (#PCDATA)>
  <!NOTATION bin SYSTEM "application/octet-stream">
  <!ENTITY blob SYSTEM "missing.bin" NDATA bin>
  <!ATTLIST response code (ok|error) #REQUIRED generated CDATA #FIXED "parser">
  <!ATTLIST row number CDATA #REQUIRED>
  <!ATTLIST cell kind (text|binary) "text" ref ENTITY #IMPLIED>
]>
<response code="ok">
  <status>ready</status>
  <data>
    <row number="01">
      <cell kind="text">first</cell>
      <cell kind="binary" ref="blob"/>
    </row>
    <row number="02"><cell>second</cell></row>
  </data>
</response>
