```xml
<?xml version="1.0"?>
<!DOCTYPE package [
  <!NOTATION gif SYSTEM "image/gif">
  <!NOTATION bin PUBLIC "-//EXAMPLE//NOTATION Binary//EN" "application/octet-stream">
  <!ENTITY logo SYSTEM "assets/logo.gif" NDATA gif>
  <!ENTITY payload SYSTEM "payload.bin" NDATA bin>
  <!ELEMENT package (manifest,asset*)>
  <!ELEMENT manifest EMPTY>
  <!ELEMENT asset EMPTY>
  <!ATTLIST package kind NOTATION (gif|bin) "gif">
  <!ATTLIST manifest main ENTITY #IMPLIED links IDREFS #IMPLIED>
  <!ATTLIST asset id ID #REQUIRED file ENTITY #REQUIRED>
]>
<package kind="gif">
  <manifest main="logo" links="logo data"/>
  <asset id="logo" file="logo"/>
  <asset id="data" file="payload"/>
</package>
```
