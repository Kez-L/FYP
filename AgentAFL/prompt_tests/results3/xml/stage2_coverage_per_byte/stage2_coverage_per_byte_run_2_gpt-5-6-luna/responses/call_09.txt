  <!ENTITY % fields "<!ELEMENT meta (field*)><!ELEMENT field (#PCDATA)>">

  <!ELEMENT part (#PCDATA)>

  <!ATTLIST field name NMTOKEN #REQUIRED type CDATA #IMPLIED>
  <!ATTLIST chunk n NMTOKEN #REQUIRED>
  <!ATTLIST separator kind CDATA #FIXED "record">

    <!ELEMENT ignored EMPTY>
    <!ATTLIST ignored never CDATA #REQUIRED>
  ]]>
]>
<packet mode="encoded">
  <meta>

  </meta>

    <chunk n="001">abc<part>DEF</part>ghi</chunk>
    <separator/>
    <chunk n="002"><part>jkl</part><![CDATA[mno]]></chunk>
  </body>
