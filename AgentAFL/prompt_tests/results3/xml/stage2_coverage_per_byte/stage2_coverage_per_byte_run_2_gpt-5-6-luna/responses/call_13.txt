```xml
<!DOCTYPE package [
  <!ENTITY % common '
    <!ELEMENT package (manifest, content)>
    <!ELEMENT manifest (file*)>
    <!ELEMENT file EMPTY>
    <!ATTLIST file path CDATA #REQUIRED kind (text|binary) "text">
    <!ELEMENT content (section|separator)*>
    <!ELEMENT section (#PCDATA|line)*>
    <!ELEMENT line EMPTY>
    <!ELEMENT separator EMPTY>
  '>
  %common;
  <!ENTITY % optional "INCLUDE">
  <![%optional;[
    <!ATTLIST package revision CDATA #IMPLIED>
    <!ELEMENT note (#PCDATA)>
  ]]>
  <!NOTATION gif SYSTEM "image/gif">
  <!ENTITY logo SYSTEM "logo.gif" NDATA gif>
]>
<package revision="r&amp;4">
  <manifest>
    <file path="readme.txt"/>
    <file path="image.gif" kind="binary"/>
  </manifest>
  <content>
    <section>first<line/>second<line/></section>
    <separator/>
    <section><![CDATA[a<b>c]]></section>
  </content>
</package>
```
