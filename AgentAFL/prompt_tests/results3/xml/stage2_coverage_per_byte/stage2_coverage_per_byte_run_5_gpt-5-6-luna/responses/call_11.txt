  <!ELEMENT description (#PCDATA|em|br)*>

]>

<!DOCTYPE matrix [
  <!ELEMENT matrix (row*)>
  <!ELEMENT row (cell*)>
  <!ELEMENT cell (#PCDATA)>
  <!ATTLIST matrix width NMTOKEN #IMPLIED>
  <!ATTLIST row index NMTOKEN #REQUIRED>
  <!ATTLIST cell kind NMTOKENS #IMPLIED span ENTITY #IMPLIED>
  <!NOTATION bin SYSTEM "application/octet-stream">
  <!ENTITY blob SYSTEM "data.bin" NDATA bin>
]>
<matrix width="03">
  <row index="0">
    <cell kind="numeric integer">1</cell>
    <cell span="blob">2</cell>
  </row>
  <row index="1"><cell>3</cell><cell kind="">4</cell></row>
</matrix>

<!DOCTYPE network [

  <!ELEMENT node EMPTY>
  <!ELEMENT edge EMPTY>
  <!ATTLIST network xmlns CDATA #IMPLIED>
  <!ATTLIST node key ID #REQUIRED class CDATA #FIXED "router">
  <!ATTLIST edge from IDREF #REQUIRED to IDREFS #REQUIRED weight NMTOKEN "0">
  <!ENTITY amp "&amp;">
]>
<network xmlns="urn:graph">
  <node key="n0" class="router"/>
  <node key="n1" class="router"/>
  <edge from="n0" to="n1 n2" weight="1&amp;2"/>
  <edge from="missing" to="n0" weight="&amp;amp;"/>
</network>
