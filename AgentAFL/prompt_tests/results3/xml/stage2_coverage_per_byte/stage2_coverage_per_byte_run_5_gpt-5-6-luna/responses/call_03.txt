<!DOCTYPE message [
  <!ELEMENT message (header, body, footer?)>
  <!ELEMENT header (from, to, subject)>

  <!ELEMENT to (#PCDATA)>

  <!ELEMENT line (#PCDATA)>
  <!ELEMENT quote (#PCDATA|line)*>
  <!ELEMENT attachment EMPTY>
  <!ATTLIST attachment

  <!ELEMENT footer (#PCDATA)>
  <!ENTITY amp "&amp;">
  <!ENTITY quoted "She said &quot;parse this&quot;">
]>
<message>
  <header>
    <from>sender@example.test</from>
    <to>receiver@example.test</to>
    <subject>&quoted;</subject>
  </header>
  <body>
    <line>One &amp; two &amp; three.</line>
    <quote>
      <line>Nested quotation</line>
      <line>Another line</line>
    </quote>
    <attachment type="binary" checksum="00ff11"/>
  </body>
  <footer>End &amp; done</footer>
</message>
