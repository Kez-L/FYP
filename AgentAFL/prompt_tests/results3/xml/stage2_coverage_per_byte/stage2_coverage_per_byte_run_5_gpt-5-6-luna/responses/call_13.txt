  <!ELEMENT section (metric+, table?)>
  <!ELEMENT metric EMPTY>
  <!ELEMENT table (row+)>
  <!ELEMENT row (cell+)>
  <!ELEMENT cell (#PCDATA)>
  <!ELEMENT warning (#PCDATA)>
  <!ATTLIST report generated %flags; #IMPLIED>
  <!ATTLIST section id ID #IMPLIED>
  <!ATTLIST metric name CDATA #REQUIRED unit CDATA #IMPLIED value CDATA #REQUIRED>
  <!ATTLIST table format (csv|grid|raw) "raw">
  <!ATTLIST row n NMTOKEN #IMPLIED>
  <!ATTLIST cell span NMTOKEN "1">
]>
<report generated="yes">
  <section id="cpu">
