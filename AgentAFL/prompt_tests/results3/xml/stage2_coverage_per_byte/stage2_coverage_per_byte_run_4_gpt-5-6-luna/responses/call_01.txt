  <!ELEMENT payload (entry+)>

  <!ATTLIST entry key ID #REQUIRED>
]>
<envelope>
  <header>&nested;</header>
  <payload>
    <entry key="e1">first &amp; second</entry>
    <entry key="e2">value &#x1F600;</entry>
  </payload>
</envelope>
