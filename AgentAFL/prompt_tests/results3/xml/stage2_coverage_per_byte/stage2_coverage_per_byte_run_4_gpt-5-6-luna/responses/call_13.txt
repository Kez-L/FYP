<?xml version="1.0"?>
<!DOCTYPE archive [
  <!NOTATION gif SYSTEM "image/gif">
  <!NOTATION jpeg PUBLIC "-//IMAGE/JPEG//" "image/jpeg">
  <!ENTITY cover SYSTEM "cover.gif" NDATA gif>
  <!ENTITY preview SYSTEM "preview.jpg" NDATA jpeg>
  <!ELEMENT archive (metadata, asset+)>
  <!ELEMENT metadata (name, checksum)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT checksum (#PCDATA)>
  <!ELEMENT asset EMPTY>
  <!ATTLIST archive version NMTOKEN #REQUIRED>
  <!ATTLIST asset id ID #REQUIRED
                 source ENTITY #IMPLIED
                 preview ENTITY #IMPLIED
                 role (cover|thumbnail|original) "original">
]>
<archive version="3.0">
  <metadata>
    <name>sample</name>
    <checksum>00ff&amp;11ee</checksum>
  </metadata>
  <asset id="a1" source="cover" role="cover"/>
  <asset id="a2" source="preview" preview="preview" role="thumbnail"/>
</archive>
