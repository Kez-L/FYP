  <!ELEMENT code (#PCDATA)>

]>
<catalog>
  <entry id="first">
    <title>Binary &amp; Text</title>
    <description>use <em>emphasis</em> and <code>code</code> here</description>
    <link href="next.xml" rel="next"/>
  </entry>
  <entry id="second" ref="first">
    <title>Picture &logo;</title>
    <link href="previous.xml" rel="prev"/>
  </entry>
</catalog>

  <!ELEMENT label (#PCDATA)>
  <!ELEMENT property (#PCDATA)>
  <!ELEMENT edge EMPTY>

  <!ENTITY quoted "&quot;quoted&quot;">
]>

  <node key="n1">
    <label>root</label>
    <property type="text">A &quoted; value</property>

    <label><![CDATA[second node]]></label>
  </node>
  <edge from="n1" to="n2" weight="0"/>

</graph>
