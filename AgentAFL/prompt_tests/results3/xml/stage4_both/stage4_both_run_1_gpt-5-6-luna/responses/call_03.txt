  <!ELEMENT label (#PCDATA)>

  <!ATTLIST task id ID #REQUIRED>

  <!ELEMENT command (#PCDATA)>
  <!ELEMENT result (#PCDATA|status)*>
  <!ELEMENT status (#PCDATA)>
]>

  <variables>
    <variable name="workspace">/var/tmp/build</variable>
    <variable name="mode">strict</variable>
  </variables>

    <label>Compilation</label>
    <task id="prepare" owner="builder">
      <name>Prepare sources</name>
      <command>mkdir -p /var/tmp/build</command>
      <result><status>complete</status> &open;</result>
    </task>
    <task id="compile-main" priority="urgent" depends="prepare">
      <name>Compile main target</name>
      <command>make all</command>
    </task>
  </phase>
  <phase id="package">
    <label>Packaging</label>

      <name>Create archive</name>
