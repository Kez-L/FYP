<?xml version="1.0"?>

  <!ELEMENT assembly (module+, summary)>
  <!ATTLIST assembly id ID #REQUIRED>
  <!ELEMENT module (header, component*)>
  <!ATTLIST module code NMTOKEN #REQUIRED depends IDREFS #IMPLIED>
  <!ELEMENT header (#PCDATA)>
  <!ELEMENT component (part, quantity, options?)>
  <!ATTLIST component state (installed|pending|removed) #REQUIRED>
  <!ELEMENT part (#PCDATA)>
  <!ELEMENT quantity (#PCDATA)>
  <!ELEMENT options (option*)>
  <!ELEMENT option EMPTY>
  <!ATTLIST option name CDATA #REQUIRED value CDATA #IMPLIED>
  <!ELEMENT summary (#PCDATA)>
]>
<assembly id="a-main">

      <quantity>1</quantity>
    </component>

      <part>fastener</part>
      <quantity>24</quantity>
