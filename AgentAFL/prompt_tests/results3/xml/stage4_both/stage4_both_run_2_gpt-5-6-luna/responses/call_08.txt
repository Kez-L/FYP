<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE workflow [
  <!ELEMENT workflow (stage+)>
  <!ELEMENT stage (task+, transition*)>
  <!ELEMENT task (input*, action, output*)>
  <!ELEMENT input EMPTY>
  <!ELEMENT action (#PCDATA|script)*>
  <!ELEMENT script (#PCDATA)>
  <!ELEMENT output EMPTY>
  <!ELEMENT transition EMPTY>
  <!ATTLIST workflow name CDATA #REQUIRED>
  <!ATTLIST stage id ID #REQUIRED mode (serial|parallel) "serial">
  <!ATTLIST task id ID #REQUIRED timeout CDATA #IMPLIED>
  <!ATTLIST input name CDATA #REQUIRED required (true|false) "true">
  <!ATTLIST action language (shell|python|text) "text">
  <!ATTLIST output name CDATA #REQUIRED path CDATA #IMPLIED>
  <!ATTLIST transition to IDREF #REQUIRED when CDATA #IMPLIED>
]>

  <stage id="prepare" mode="parallel">
    <task id="fetch" timeout="30s">
      <input name="remote"/>
      <action language="shell">fetch --quiet <script>source</script></action>
      <output name="workspace" path="/tmp/work"/>
    </task>
