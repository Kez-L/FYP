<!DOCTYPE catalog [
  <!ELEMENT catalog (entry*)>
  <!ELEMENT entry (name, metadata, payload)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT metadata (tag*)>
  <!ELEMENT tag EMPTY>
  <!ATTLIST tag
    key CDATA #REQUIRED
    value CDATA #IMPLIED>
  <!ELEMENT payload ANY>
  <!ENTITY % commonAttrs ' xml:id ID #IMPLIED xml:base URI #IMPLIED '>
  <!ATTLIST entry %commonAttrs; status (new|old|archived) "new">
  <!NOTATION bin SYSTEM "application/octet-stream">
  <!ENTITY data SYSTEM "data.bin" NDATA bin>
]>
<catalog>
  <entry xml:id="e-1" xml:base="urn:base/" status="archived">
    <name>Binary reference</name>
    <metadata>
      <tag key="format" value="bin"/>
      <tag key="empty"/>
    </metadata>
    <payload>&data;</payload>
  </entry>
</catalog>
