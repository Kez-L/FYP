  <!ATTLIST property key NMTOKEN #REQUIRED value CDATA #IMPLIED>

  <!NOTATION binary SYSTEM "application/octet-stream">
  <!ENTITY archive SYSTEM "archive.bin" NDATA binary>
]>

    <name>sample</name>
    <checksum>sha256:0000000000000000</checksum>

    <property key="owner"/>
  </metadata>

  <!ENTITY close "]" >
