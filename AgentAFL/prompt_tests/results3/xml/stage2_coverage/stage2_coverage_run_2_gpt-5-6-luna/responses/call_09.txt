  <!ELEMENT value (#PCDATA)>

  <!ATTLIST metadata key CDATA #REQUIRED value CDATA #IMPLIED>
  <!NOTATION bin SYSTEM "application/octet-stream">
  <!ENTITY sample "line one&#10;line two &amp; more">
]>
<map>
  <metadata key="format" value="graph"/>

    <label>Root</label>
    <property name="note"><value>&sample;</value></property>

  <node id="n2" kind="isolated"/>

<?xml version="1.0"?>
<!DOCTYPE configuration [
  <!ELEMENT configuration (defaults,profile+)>
  <!ELEMENT defaults EMPTY>
  <!ATTLIST defaults retries NMTOKEN "3" timeout CDATA "30s" enabled (true|false) "true">
  <!ELEMENT profile (inherits?,setting*,path*)>
  <!ATTLIST profile name NMTOKEN #REQUIRED>
  <!ELEMENT inherits EMPTY>
  <!ATTLIST inherits from NMTOKEN #IMPLIED>
  <!ELEMENT setting (#PCDATA)>
  <!ATTLIST setting key NMTOKEN #REQUIRED type (string|integer|boolean|list) "string">
  <!ELEMENT path (#PCDATA)>
  <!ATTLIST path name NMTOKEN #REQUIRED mode (read|write|execute) "read">
]>
<configuration>
  <defaults retries="0" timeout="PT0S" enabled="false"/>
  <profile name="base">
    <setting key="message">A &lt; B &amp;&amp; C &gt; D</setting>
    <setting key="workers" type="integer">0008</setting>
    <path name="cache" mode="write">/var/tmp/cache/</path>
  </profile>
  <profile name="child">
    <inherits from="base"/>
    <setting key="flags" type="list">alpha,beta,,gamma</setting>
    <setting key="enabled" type="boolean">TRUE</setting>
    <path name="logs" mode="execute">relative/../logs</path>
  </profile>

<?xml version="1.0"?>
<!DOCTYPE archive [
  <!ELEMENT archive (record*)>
  <!ELEMENT record (key,content,checksum?)>
  <!ATTLIST record category NMTOKEN #IMPLIED timestamp CDATA #IMPLIED>
  <!ELEMENT key (#PCDATA)>
  <!ELEMENT content (text|binary|fragment)*>
  <!ELEMENT text (#PCDATA|mark)*>
  <!ELEMENT mark (#PCDATA)>
  <!ATTLIST mark type (insert|delete|replace) #REQUIRED>
  <!ELEMENT binary EMPTY>
  <!ATTLIST binary encoding (base64|hex) #REQUIRED length NMTOKEN #IMPLIED>
  <!ELEMENT fragment ANY>
  <!ELEMENT checksum (#PCDATA)>
  <!ENTITY left "[" >
  <!ENTITY right "]" >
]>
<archive>
  <record category="text" timestamp="2026-09-24T00:00:00Z">
    <key>alpha</key>
    <content>
      <text>before <mark type="insert">new</mark> after</text>
      <fragment><unknown xmlns="urn:foreign"><x>raw</x></unknown></fragment>
    </content>
    <checksum>sha256:0000</checksum>
  </record>
  <record category="binary">
    <key>&left;payload&right;</key>
    <content><binary encoding="base64" length="0"/></content>
  </record>
</archive>
