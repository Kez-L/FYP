  <!ELEMENT catalog (section+)>

  <!ATTLIST section
    name NMTOKEN #REQUIRED
    active (yes|no) "yes">

  <!ATTLIST item
    id ID #REQUIRED
    ref IDREFS #IMPLIED
    kind (alpha|numeric|mixed) "mixed">
  <!ELEMENT label (#PCDATA)>
  <!ELEMENT value (#PCDATA|emph)*>
  <!ELEMENT emph (#PCDATA)>
  <!ELEMENT note (#PCDATA)>
  <!ATTLIST note %common;>
]>
<catalog>
  <section name="main">
    <item id="i1" kind="alpha">
      <label>A</label>
      <value>left<emph>middle</emph>right</value>
    </item>
    <item id="i2" ref="i1" kind="numeric">
      <label>42</label>
      <value>1&amp;2&lt;3</value>
    </item>
    <note>tail</note>
  </section>
</catalog>

  <!ELEMENT include EMPTY>
  <!ATTLIST include
    href URI #REQUIRED
    mode (merge|replace) "merge">

  <!ATTLIST setting
    name NMTOKEN #REQUIRED
    value CDATA #REQUIRED
    locked (true|false) "false">
  <!ELEMENT choice (option, option?)>
  <!ELEMENT option (#PCDATA)>
  <!ATTLIST option selected (0|1) "0">
  <!NOTATION bin SYSTEM "application/octet-stream">
  <!ENTITY blob SYSTEM "data.bin" NDATA bin>
]>
<config>
  <include href="file:///tmp/example.xml" mode="replace"/>
  <setting name="path" value="/a/b/c" locked="true"/>
  <setting name="quote" value="&quot;&apos;&gt;&lt;" />
  <choice>
    <option selected="1">first</option>
    <option>second</option>
  </choice>
</config>
