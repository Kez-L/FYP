<?xml version="1.0"?>
<!DOCTYPE config [
  <!ELEMENT config (defaults, profile+)>
  <!ELEMENT defaults (option*)>
  <!ELEMENT profile (inherits?, option*, path*)>
  <!ELEMENT inherits EMPTY>
  <!ELEMENT option (#PCDATA)>
  <!ELEMENT path (#PCDATA)>
  <!ATTLIST profile name ID #REQUIRED>
  <!ATTLIST option key NMTOKEN #REQUIRED enabled (true|false) "true">
  <!ATTLIST path kind (input|output|cache) #REQUIRED>
  <!ATTLIST inherits ref IDREF #REQUIRED>
  <!ENTITY nl "&#10;">
  <!ENTITY tab "&#9;">
]>
<config>
  <defaults>
    <option key="verbose" enabled="false">off</option>
    <option key="retries">3</option>
  </defaults>
  <profile name="base">
    <option key="format">xml&nl;compact</option>
    <path kind="cache">/var/tmp/app</path>
  </profile>
  <profile name="child">
    <inherits ref="base"/>
    <option key="format">pretty&tab;utf8</option>
    <option key="escaped">a&lt;b&gt;c&amp;d</option>
    <path kind="input">./data/</path>
    <path kind="output">./out/</path>
  </profile>
</config>

<!DOCTYPE matrix [
  <!ELEMENT matrix (row+)>
  <!ELEMENT row (cell+)>
  <!ELEMENT cell (#PCDATA|value)*>
  <!ELEMENT value (#PCDATA)>
  <!ATTLIST matrix units CDATA #IMPLIED>
  <!ATTLIST row index NMTOKEN #REQUIRED>
  <!ATTLIST cell type (text|number|empty) "text" colspan NMTOKEN "1">
  <!ENTITY left "(">
  <!ENTITY right ")">
]>
<matrix units="mixed">
  <row index="0">
    <cell type="text">name</cell>
    <cell type="number">42</cell>
    <cell type="empty"/>
  </row>
  <row index="1">
    <cell colspan="2">alpha <value>&left;beta&right;</value></cell>
  </row>
  <row index="2">
    <cell type="text">line one&#x0A;line two</cell>
    <cell type="number">-0.000</cell>
    <cell type="text"><![CDATA[<unparsed>&still raw;</unparsed>]]></cell>
