  <!ENTITY close "</section>">

  <!ELEMENT report (meta, section*)>
  <!ELEMENT meta EMPTY>
  <!ATTLIST meta generated CDATA #REQUIRED>
  <!ELEMENT section (heading, paragraph*)>
  <!ATTLIST section number NMTOKEN #REQUIRED>
  <!ELEMENT heading (#PCDATA)>
  <!ELEMENT paragraph (#PCDATA|code|link)*>
  <!ELEMENT code (#PCDATA)>
  <!ELEMENT link (#PCDATA)>
  <!ATTLIST link target CDATA #REQUIRED>
]>
<report>
  <meta generated="2026-10-01"/>
  <section number="1">
    <heading>Introduction &mark;</heading>
    <paragraph>Use <code>&lt;tag&gt;</code> or <link target="urn:ref">this link</link>.</paragraph>
  </section>
  <section number="2">
    <heading>Details</heading>
