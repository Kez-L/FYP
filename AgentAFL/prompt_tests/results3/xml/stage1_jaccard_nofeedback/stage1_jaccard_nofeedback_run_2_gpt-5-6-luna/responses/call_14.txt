  <!ELEMENT section (title, entry*)>
  <!ATTLIST section key ID #REQUIRED>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT entry (value, link?)>
  <!ATTLIST entry ref IDREF #IMPLIED kind CDATA #IMPLIED>
  <!ELEMENT value (#PCDATA)>
  <!ELEMENT link EMPTY>
  <!ATTLIST link href URI #REQUIRED>
  <!NOTATION binary SYSTEM "application/octet-stream">
]>
<report>

      <value>left &lt; middle &gt; right</value>
    </entry>
  </section>

  </section>
</report>
