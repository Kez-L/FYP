  <!ELEMENT archive (header, record+)>
  <!ELEMENT header (#PCDATA)>
  <!ELEMENT record (title, body, metadata?)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT body (#PCDATA|em|ref)*>
  <!ELEMENT em (#PCDATA)>
  <!ELEMENT ref EMPTY>
  <!ELEMENT metadata (tag*)>
  <!ELEMENT tag (#PCDATA)>
  <!ATTLIST record id ID #REQUIRED status (new|old|held) "new">
  <!ATTLIST ref target IDREF #IMPLIED>
  <!ATTLIST tag key CDATA #REQUIRED>
  <!ENTITY mark "&#x2605;">
]>
<archive xmlns="urn:example:archive" xmlns:x="urn:example:meta">
  <header>Entries &mark; and <em>notes</em>.</header>
  <record id="r1" status="new">
    <title><![CDATA[An <unparsed> title]]></title>
    <body>Before <em>important</em> text <ref target="r1"/> after.</body>
    <metadata>
      <tag key="owner">delta</tag>
      <tag x:class="priority" key="rank">7</tag>
    </metadata>
  </record>
</archive>

<?xml version="1.0"?>
<!DOCTYPE catalog [
  <!ELEMENT catalog (section*)>
  <!ELEMENT section (item|section)*>
  <!ELEMENT item EMPTY>
  <!ATTLIST section name CDATA #IMPLIED>
  <!ATTLIST item code NMTOKEN #REQUIRED>
  <!NOTATION bin SYSTEM "application/octet-stream">
  <!ENTITY payload SYSTEM "blob.bin" NDATA bin>
]>
<catalog>
  <section name="outer">
    <item code="A-01"/>
    <section name="inner">
      <item code="B_02"></item>
      <item code="C.03"/>
    </section>
  </section>
  <section>
    <!-- deliberately adjacent empty elements -->
    <item code="Z9"/>
    <?catalog-process mode="compact"?>
  </section>
</catalog>

<!DOCTYPE envelope [

  %switch;
  <!ELEMENT meta EMPTY>
  <!ELEMENT payload (chunk+)>
  <!ELEMENT chunk (#PCDATA|part)*>
  <!ELEMENT part (#PCDATA)>
  <!ELEMENT signature (#PCDATA)>
  <!ATTLIST meta timestamp CDATA #IMPLIED>
  <!ATTLIST chunk n CDATA #REQUIRED>
  <!ENTITY left "[" >
  <!ENTITY right "]" >
]>
<envelope>
  <meta timestamp="1970-01-01T00:00:00Z"/>
