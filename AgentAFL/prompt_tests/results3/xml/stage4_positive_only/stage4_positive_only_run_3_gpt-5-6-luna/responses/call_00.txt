<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE report [
  <!ENTITY % switches "<!ENTITY draft 'temporary'><!ATTLIST report mode (full|delta) 'full'>">
  %switches;
  <!ELEMENT report (title, group+)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT group (record+)>
  <!ATTLIST group name CDATA #REQUIRED>
  <!ELEMENT record (key, value, note?)>
  <!ATTLIST record active (yes|no) 'yes'>
  <!ELEMENT key (#PCDATA)>
  <!ELEMENT value (#PCDATA)>
  <!ELEMENT note (#PCDATA)>
  <![IGNORE[
    <!ELEMENT ignored (anything)>
    <!ENTITY hidden "not expanded">
  ]]>
]>
<report mode="delta">
  <title>Quarterly &draft; report</title>
  <group name="north">
    <record active="yes"><key>k1</key><value>42</value><note><![CDATA[x < y]]></note></record>
    <record><key>k2</key><value>7</value></record>
  </group>
</report>
