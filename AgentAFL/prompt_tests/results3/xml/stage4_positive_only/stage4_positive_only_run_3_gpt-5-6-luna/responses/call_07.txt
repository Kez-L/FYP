    <var name="REGION" value="eu-central" secret="no"/>
    <var name="TOKEN" secret="yes"/>
    <var name="RETAIN_DAYS" value="30"/>
  </variables>

    <stage id="prepare">
      <task id="scan" executor="filesystem">
        <input name="root">/srv/data/${REGION}</input>

        <input name="key">env:TOKEN</input>
        <output name="signature">/var/archive/nightly.tar.zst.sig</output>
      </task>
      <on-success>notify</on-success>
      <on-failure>retry-package</on-failure>
    </stage>
  </stages>
