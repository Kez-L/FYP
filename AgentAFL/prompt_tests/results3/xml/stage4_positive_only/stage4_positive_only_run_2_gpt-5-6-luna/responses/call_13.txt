    <input name="endpoint" value="https://example.invalid/data?mode=full&amp;limit=20"/>
    <action>download</action>
    <output name="response" path="/var/tmp/input.json"/>
  </step>
  <step id="transform" depends-on="fetch">
    <input name="format" value="normalized"/>
    <action><![CDATA[
      select records
      where status != "deleted"
    ]]></action>

  </step>
