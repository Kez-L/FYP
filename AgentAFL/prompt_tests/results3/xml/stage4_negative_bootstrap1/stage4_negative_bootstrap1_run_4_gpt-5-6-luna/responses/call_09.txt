  <!ELEMENT variables (var*)>
  <!ELEMENT var EMPTY>
  <!ATTLIST var name ID #REQUIRED value CDATA #IMPLIED>
  <!ELEMENT steps (step+)>
  <!ELEMENT step (input*, output*)>
  <!ATTLIST step id ID #REQUIRED after IDREFS #IMPLIED state (pending|ready|done) "pending">

  <!ELEMENT output (#PCDATA)>
]>

  <variables>
    <var name="source" value="a&#x85;b"/>

      <input ref="source"/>
      <output>ready&#x2028;for processing</output>
    </step>
    <step id="transform" after="prepare">
      <input ref="threshold"/>
      <output>result&#x0D;&#x0A;pending</output>
    </step>
    <step id="publish" after="prepare transform" state="ready">
      <input ref="source"/>
      <output>destination: archive</output>
    </step>

</workflow>
