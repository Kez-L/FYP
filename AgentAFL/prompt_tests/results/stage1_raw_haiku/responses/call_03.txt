```xml
<!DOCTYPE mixed [
  <!ENTITY internal "nested">
  <!NOTATION gif PUBLIC "image/gif">
  <!ELEMENT mixed (#PCDATA|branch)*>
  <!ELEMENT branch (#PCDATA)>
  <!ATTLIST branch type (A|B|C) "A">
]>
<mixed>
  <branch type="B">
    Content &internal; here
  </branch>
  <branch type="C"/>
  Mixed text &internal;
</mixed>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE catalog [
  <!ENTITY % remote SYSTEM "http://example.com/data.xml">
  <!ENTITY local "replacement">
  <!ELEMENT catalog (item*)>
  <!ELEMENT item (name, price)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT price (#PCDATA)>
  <!ATTLIST item sku CDATA #FIXED "SKU123">
]>
<catalog>
  <item sku="SKU123">
    <name>&local;</name>
    <price>99.99</price>
  </item>
</catalog>
```
