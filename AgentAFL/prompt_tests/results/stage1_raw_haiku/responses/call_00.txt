```xml
<?xml version="1.0"?>
<!DOCTYPE root [
  <!ELEMENT root (child*)>
  <!ELEMENT child (#PCDATA)>
  <!ATTLIST child id ID #REQUIRED>
  <!ENTITY internal "recursive">
]>
<root>
  <child id="c1">&internal;</child>
  <child id="c2">text</child>
</root>
```

```xml

  <!ELEMENT section (param*)>
  <!ELEMENT param (#PCDATA)>
  <!ATTLIST section name CDATA #REQUIRED>
  <!ATTLIST param key CDATA #REQUIRED>
]>
<config>
  <section name="first">
    <param key="a">value1</param>
    <param key="b">value2</param>
  </section>
  <section name="second">
    <param key="c">value3</param>
  </section>
</config>
```
