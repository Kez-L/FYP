<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE message [
  <!ELEMENT message (header, body, footer?)>
  <!ELEMENT header (from, to, subject)>
  <!ELEMENT body (#PCDATA|emphasis|linebreak)*>
  <!ELEMENT footer (#PCDATA)>
  <!ELEMENT from (#PCDATA)>
  <!ELEMENT to (#PCDATA)>
  <!ELEMENT subject (#PCDATA)>
  <!ELEMENT emphasis (#PCDATA)>
  <!ELEMENT linebreak EMPTY>
  <!ATTLIST message priority (low|normal|high) "normal">
  <!ENTITY smile ":-)">
]>
<message priority="high">
  <header>
    <from>Alice</from>
    <to>Bob</to>
    <subject>Test &smile;</subject>
  </header>
  <body>
    Hello <emphasis>world</emphasis><linebreak/>
    Second line with &#13; carriage return
  </body>
  <footer>sent</footer>
</message>
