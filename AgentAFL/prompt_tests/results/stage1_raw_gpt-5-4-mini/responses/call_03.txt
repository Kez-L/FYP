  <!ELEMENT r (a, b?, c*)>

```xml
<?xml version="1.0"?>
<!DOCTYPE catalog [
  <!ELEMENT catalog (book+)>
  <!ELEMENT book (title, author, meta?)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT author (#PCDATA)>
  <!ELEMENT meta EMPTY>
  <!ATTLIST book id ID #REQUIRED>
  <!ATTLIST book lang CDATA "en">
]>
<catalog>
  <book id="b1">
    <title>Alpha</title>
    <author>Ada</author>
    <meta/>
  </book>
  <book id="b2" lang="fr">
    <title>Beta</title>
    <author>Bob</author>
  </book>
</catalog>
```

]>

    <node>n1</node>
    <leaf/>

```xml
<?xml version="1.0"?>
<!DOCTYPE data [
  <!ELEMENT data (item+, trailer)>
  <!ELEMENT item (name, value)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT value (#PCDATA)>
  <!ELEMENT trailer (#PCDATA)>
  <!ATTLIST item type CDATA #IMPLIED flag (yes|no) "no">
]>
<data>
  <item type="x" flag="yes">
    <name>first</name>
    <value>10</value>
  </item>
  <item>
    <name>second</name>
    <value>20</value>
  </item>
  <trailer>done</trailer>
</data>
```
