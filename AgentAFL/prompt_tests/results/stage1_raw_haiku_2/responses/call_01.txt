```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE root [
  <!ELEMENT root (child*)>
  <!ELEMENT child (#PCDATA)>
  <!ATTLIST child id ID #REQUIRED>
  <!ENTITY % param1 "test">
  <!ENTITY % param2 "%param1;%param1;%param1;">
]>
<root>
  <child id="c1">&param2;</child>
  <child id="c2">content</child>
</root>
```
