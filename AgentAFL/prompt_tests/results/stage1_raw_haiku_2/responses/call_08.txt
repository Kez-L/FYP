<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE root [
  <!ENTITY % param1 "value1">
  <!ENTITY % param2 "<!ENTITY intern '%param1;'>">

]>

  <elem>&intern;</elem>
</root>

  <!ENTITY test "&#38;#48;&#48;&#48;;">
]>
