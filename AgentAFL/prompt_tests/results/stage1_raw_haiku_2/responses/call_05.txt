```xml
<?xml version="1.0"?>
<!DOCTYPE doc [
  <!NOTATION gif SYSTEM "image/gif">
  <!ELEMENT doc (item*)>
  <!ELEMENT item (#PCDATA)>
  <!ATTLIST item type NOTATION (gif) #IMPLIED>
]>
<doc>
  <item type="gif">data</item>
</doc>
```
