```xml
<?xml version="1.0"?>
<!DOCTYPE doc [
  <!ELEMENT doc (section*)>
  <!ELEMENT section (title, para+)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT para (#PCDATA|ref)*>
  <!ELEMENT ref EMPTY>
  <!ATTLIST ref href CDATA #REQUIRED>
  <!ENTITY copy "&#169;">
  <!ENTITY % rights "Copyright &copy; 2024">
]>
<doc>
  <section>
    <title>Section One</title>
    <para>Text with &rights; <ref href="http://example.com"/></para>
  </section>
</doc>
```
