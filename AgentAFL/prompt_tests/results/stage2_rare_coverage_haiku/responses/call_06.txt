```xml
<?xml version="1.0"?>
<!DOCTYPE doc [
<!ELEMENT doc (section)*>
<!ELEMENT section (title, para+)>
<!ELEMENT title (#PCDATA)>
<!ELEMENT para (#PCDATA | emphasis | strong)*>
<!ELEMENT emphasis (#PCDATA)>
<!ELEMENT strong (#PCDATA)>
<!ATTLIST section id ID #REQUIRED level (1|2|3) "1">
]>
<doc>
  <section id="sec1" level="1">
    <title>Introduction</title>
    <para>This is <emphasis>important</emphasis> text.</para>
    <para>This is <strong>very strong</strong> content.</para>
  </section>
  <section id="sec2" level="2">
    <title>Details</title>
    <para>More <emphasis>content</emphasis> with <strong>mixed</strong> formatting.</para>
  </section>
</doc>
```
