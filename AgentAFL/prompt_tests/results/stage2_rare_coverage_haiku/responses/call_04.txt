```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE root [
<!ELEMENT root (item+)>
<!ELEMENT item (#PCDATA)>
<!ATTLIST item id ID #REQUIRED>
<!ATTLIST item ref IDREF #IMPLIED>
]>
<root>
  <item id="i1">text1</item>
  <item id="i2" ref="i1">text2</item>
  <item id="i3" ref="i2">text3</item>
</root>
```

```xml
<!DOCTYPE doc [
<!ELEMENT doc (section*)>
<!ELEMENT section (para+)>
<!ELEMENT para (#PCDATA | emphasis)*>
<!ELEMENT emphasis (#PCDATA)>
<!ATTLIST para id CDATA #IMPLIED>
]>
<doc>
  <section>
    <para id="p1">First paragraph with <emphasis>emphasized</emphasis> text</para>
    <para id="p2">Second paragraph</para>
  </section>
  <section>
    <para id="p3">Third paragraph</para>
  </section>
</doc>
```

```xml

<!ELEMENT entry EMPTY>
<!ATTLIST entry
  a CDATA #REQUIRED
  b CDATA #REQUIRED
  c (x|y|z) "x"
  d NMTOKEN #IMPLIED
  e NMTOKENS #IMPLIED
>
]>
<test>
  <entry a="val_a_1" b="val_b_1" c="x" d="token1" e="tok1 tok2"/>
  <entry a="val_a_2" b="val_b_2" c="y" d="token2"/>
  <entry a="val_a_3" b="val_b_3" c="z"/>
```
