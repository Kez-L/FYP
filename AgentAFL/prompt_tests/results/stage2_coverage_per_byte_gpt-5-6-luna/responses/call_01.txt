```xml
    label CDATA #IMPLIED
    refs IDREFS #IMPLIED
    icon ENTITY #IMPLIED>
  <!ELEMENT edge EMPTY>
  <!ATTLIST edge
    from IDREF #REQUIRED
    to IDREF #REQUIRED
    weight NMTOKEN "1">

  <!ENTITY unused SYSTEM "asset.png" NDATA png>
]>
<graph>
  <node id="n1" label="root" refs="n2 n3"/>
  <node id="n2" label="left"/>
  <node id="n3" label="right"/>
  <edge from="n1" to="n2" weight="0x10"/>

</graph>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE config [
  <!ENTITY % declarations "
    <!ELEMENT option EMPTY>
    <!ATTLIST option
      key CDATA #REQUIRED
      value CDATA #IMPLIED
      locked (yes|no) 'no'>
  ">
  %declarations;
  <!ELEMENT config (section*)>
  <!ELEMENT section (option*, note?)>
  <!ATTLIST section name NMTOKEN #REQUIRED>
  <!ELEMENT note (#PCDATA)>
  <!ENTITY quoted "&#x22;quoted&#x22;">
]>
<config>
  <section name="network">
    <option key="host" value="localhost"/>
    <option key="port" value="8080" locked="yes"/>
    <note>Use &quoted; values only.</note>
  </section>
  <section name="empty"/>
</config>
```
