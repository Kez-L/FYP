<?xml version="1.0"?>
<!DOCTYPE catalog [
  <!ENTITY % common " id ID #REQUIRED">
  <!ELEMENT catalog (entry+)>
  <!ELEMENT entry (title,meta?,description)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT meta EMPTY>
  <!ELEMENT description (#PCDATA|em|br)*>
  <!ELEMENT em (#PCDATA)>
  <!ELEMENT br EMPTY>
  <!ATTLIST entry %common; status (new|old|archived) "new">
  <!ATTLIST meta key CDATA #IMPLIED value CDATA #IMPLIED>
  <!NOTATION bin SYSTEM "application/octet-stream">
  <!ENTITY payload SYSTEM "data.bin" NDATA bin>
]>
<catalog>
  <entry id="a1" status="old">
    <title>First &amp; Second</title>
    <meta key="format" value="bin"/>
    <description>Text <em>with emphasis</em><br/>after break.</description>
  </entry>
</catalog>
