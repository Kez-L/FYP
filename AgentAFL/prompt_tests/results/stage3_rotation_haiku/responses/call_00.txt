```xml

  <!ELEMENT mixed (text|cdata)*>
  <!ELEMENT text (#PCDATA)>
  <!ELEMENT cdata ANY>
  <!ENTITY recursive "&recursive;">
]>
<mixed>
  <text>Plain text content</text>
  <cdata><![CDATA[Literal <>&"' chars]]></cdata>

</mixed>
```

```xml
  <!ENTITY copy "&#169;">
```
