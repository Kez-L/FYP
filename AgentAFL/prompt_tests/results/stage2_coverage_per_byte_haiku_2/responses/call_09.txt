```xml
  <!ELEMENT b EMPTY>
  <!ELEMENT c (#PCDATA)>
  <!ELEMENT nested (#PCDATA)>
  <!ATTLIST a id ID #REQUIRED ref IDREF #IMPLIED>
  <!ATTLIST b href CDATA #FIXED "http://test">
  <!NOTATION jpeg PUBLIC "image/jpeg">

<test>
  <a id="first">
    <nested>content</nested>
  </a>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE manifest [
  <!ELEMENT manifest (permission | activity | provider)*>
  <!ELEMENT permission EMPTY>
  <!ELEMENT activity (intent-filter*)>
  <!ELEMENT provider EMPTY>
  <!ELEMENT intent-filter (action | category)*>
  <!ELEMENT action EMPTY>
  <!ELEMENT category EMPTY>
  <!ATTLIST permission name CDATA #REQUIRED level (user|system|admin) "user">
  <!ATTLIST activity name CDATA #REQUIRED exported (true|false) #IMPLIED>
  <!ATTLIST action name CDATA #REQUIRED>
  <!ATTLIST category name CDATA #REQUIRED>
  <!ATTLIST provider name CDATA #REQUIRED authorities CDATA #IMPLIED>
]>
<manifest>
  <permission name="READ_DATA" level="system"/>
  <activity name=".MainActivity" exported="true">
    <intent-filter>
      <action name="android.intent.action.MAIN"/>
      <category name="android.intent.category.LAUNCHER"/>
    </intent-filter>
  </activity>
  <provider name=".DataProvider" authorities="com.app.data"/>
</manifest>
```
