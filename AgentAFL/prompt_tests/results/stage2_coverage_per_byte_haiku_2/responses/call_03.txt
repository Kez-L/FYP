```xml
<!DOCTYPE schema [
  <!ELEMENT schema (type+, element*)>
  <!ELEMENT type (restriction | extension)>
  <!ELEMENT restriction (pattern | length)*>
  <!ELEMENT pattern EMPTY>
  <!ELEMENT length EMPTY>
  <!ELEMENT extension (attribute*)>
  <!ELEMENT attribute EMPTY>
  <!ELEMENT element (annotation?)>
  <!ELEMENT annotation (#PCDATA)>
  <!ATTLIST type name ID #REQUIRED base CDATA #IMPLIED>
  <!ATTLIST restriction value CDATA #REQUIRED>
  <!ATTLIST pattern regex CDATA #REQUIRED>
  <!ATTLIST length min CDATA #IMPLIED max CDATA #IMPLIED>
  <!ATTLIST attribute ref CDATA #REQUIRED required (true|false) "false">
  <!ATTLIST element name CDATA #REQUIRED type CDATA #REQUIRED>
]>
<schema>
  <type name="stringType" base="xs:string">
    <restriction value="limited">
      <pattern regex="[a-z]+"/>
      <length min="1" max="255"/>
    </restriction>
  </type>
  <element name="field1" type="stringType">
    <annotation>A test field</annotation>
  </element>
</schema>
```

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE events [
  <!ELEMENT events (event)*>
  <!ELEMENT event (timestamp, source, tags?, payload)>
  <!ELEMENT timestamp (#PCDATA)>
  <!ELEMENT source EMPTY>
  <!ELEMENT tags (tag+)>
  <!ELEMENT tag (#PCDATA)>
  <!ELEMENT payload (data | binary)>
  <!ELEMENT data (#PCDATA)>
  <!ELEMENT binary EMPTY>
  <!ATTLIST source system CDATA #REQUIRED priority (low|medium|high) "medium">
  <!ATTLIST tags count NMTOKEN #IMPLIED>
  <!ATTLIST tag severity (info|warn|error) "info">
  <!ATTLIST binary encoding (base64|hex) "base64" size CDATA #REQUIRED>
]>
<events>
  <event>
    <timestamp>2024-01-15T10:30:45Z</timestamp>
    <source system="parser" priority="high"/>
    <tags count="2">
      <tag severity="warn">critical</tag>
      <tag severity="info">test</tag>
    </tags>
    <payload>
      <data>event payload content</data>
    </payload>
  </event>
</events>
```
