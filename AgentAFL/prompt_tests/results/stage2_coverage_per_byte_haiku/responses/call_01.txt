<!DOCTYPE root [
  <!ENTITY internal "expansion">
  <!ELEMENT root (child*)>

]>

  <child>&internal;</child>
  <child>text</child>
</root>

```xml
<?xml version="1.0"?>
<!DOCTYPE root [
  <!ELEMENT root (a|b)+>
  <!ELEMENT a EMPTY>
  <!ELEMENT b (#PCDATA|a)*>
  <!ATTLIST a id ID #REQUIRED>
  <!ATTLIST b ref IDREF #IMPLIED>
]>
<root>
  <a id="a1"/>
  <b ref="a1">content</b>
</root>
```

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE data [
  <!NOTATION jpeg SYSTEM "image/jpeg">
  <!ENTITY logo SYSTEM "logo.jpg" NDATA jpeg>
  <!ELEMENT data (unparsed*)>
  <!ELEMENT unparsed EMPTY>
  <!ATTLIST unparsed file ENTITY #REQUIRED>
]>
<data>
  <unparsed file="logo"/>
</data>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE root [
  <!ELEMENT root (nested)+>
  <!ELEMENT nested (nested|leaf)*>
  <!ELEMENT leaf EMPTY>
  <!ATTLIST nested level CDATA #IMPLIED xmlns:custom CDATA #FIXED "http://custom.ns">
]>
<root>
  <nested level="1" xmlns:custom="http://custom.ns">
    <nested level="2">
      <leaf/>
    </nested>
  </nested>
</root>
```

  <!ELEMENT title (#PCDATA)>
  <!ELEMENT content (#PCDATA|subsection)*>
  <!ELEMENT subsection (#PCDATA)>
]>

  <section>
    <title>First</title>
