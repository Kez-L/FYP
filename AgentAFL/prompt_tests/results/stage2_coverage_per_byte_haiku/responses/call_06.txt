```xml
  <!ENTITY intern "internal entity content">

]>

  <child>text</child>
</root>
```

```xml
]>
<data>
  <item id="id1" type="test">content</item>
  <item id="id2">nested</item>
</data>
```

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE config [
  <!ELEMENT config (setting)*>
  <!ELEMENT setting (#PCDATA)>
  <!ATTLIST setting
    name CDATA #REQUIRED
    value CDATA #IMPLIED
    enabled (true|false) "true">
  <!ENTITY % file SYSTEM "file://internal">
]>
<config>
  <setting name="timeout" value="30" enabled="true">30000ms</setting>
  <setting name="debug" enabled="false"/>
</config>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE catalog [
  <!ELEMENT catalog (product*)>
  <!ELEMENT product (name, price, quantity)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT price (#PCDATA)>
  <!ELEMENT quantity (#PCDATA)>
  <!ATTLIST product id ID #REQUIRED sku CDATA #FIXED "SKU">
  <!ATTLIST price currency CDATA "USD">
]>
<catalog>
  <product id="prod1" sku="SKU">
    <name>Item</name>
    <price currency="EUR">99</price>
    <quantity>5</quantity>
  </product>
</catalog>
```
