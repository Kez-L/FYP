```
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE root [
  <!ELEMENT root (item)*>
  <!ELEMENT item (#PCDATA)>
  <!ENTITY myent "expanded content">
  <!NOTATION gif PUBLIC "image/gif">
]>
<root>
  <item>&myent;</item>
  <item>text</item>
</root>
```

```
<?xml version="1.0"?>
<!DOCTYPE html [
  <!ELEMENT html (head, body)>
  <!ELEMENT head (title)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT body (p+)>
  <!ELEMENT p (#PCDATA|b|i)*>
  <!ELEMENT b (#PCDATA)>
  <!ELEMENT i (#PCDATA)>
]>
<html>
  <head><title>Test</title></head>
  <body><p>Hello <b>world</b></p></body>
</html>
```

```
<?xml version="1.0" encoding="UTF-16"?>
<!DOCTYPE config [
  <!ELEMENT config (setting)*>
  <!ELEMENT setting EMPTY>
  <!ATTLIST setting name CDATA #REQUIRED value CDATA #IMPLIED>
  <!ATTLIST setting type (string|int|bool) "string">
]>
<config>
  <setting name="debug" value="true" type="bool"/>
  <setting name="timeout"/>
</config>
```

```
<!DOCTYPE catalog [
  <!ELEMENT catalog (product, product)>
  <!ELEMENT product (name, price, description?)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT price (#PCDATA)>
  <!ELEMENT description (#PCDATA)>
  <!ATTLIST product id ID #REQUIRED>
  <!ATTLIST product category CDATA #FIXED "electronics">
]>
<catalog>
  <product id="p1" category="electronics">
    <name>Widget</name>
    <price>9.99</price>
  </product>
  <product id="p2">
    <name>Gadget</name>
    <price>19.99</price>
    <description>A useful tool</description>
  </product>
</catalog>
```

```
<?xml version="1.0" standalone="yes"?>
<!DOCTYPE data [
  <!ELEMENT data (record*)>
  <!ELEMENT record (field+)>
  <!ELEMENT field (#PCDATA)>
  <!ATTLIST record xmlns:custom CDATA #IMPLIED>
  <!ATTLIST field id CDATA #REQUIRED type (text|number|date) #IMPLIED>
  <!ENTITY nbsp "&#160;">
]>
<data>
  <record>
    <field id="1" type="text">Value&nbsp;One</field>
    <field id="2" type="number">42</field>
  </record>
  <record xmlns:custom="http://custom.org">
    <field id="3">Another</field>
  </record>
</data>
```
