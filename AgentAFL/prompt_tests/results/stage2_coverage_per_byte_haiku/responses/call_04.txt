```xml
<?xml version="1.0"?>
<!DOCTYPE data [
  <!ELEMENT data (section*, footer?)>
  <!ELEMENT section (#PCDATA | subsection)*>
  <!ELEMENT subsection (#PCDATA)>
  <!ELEMENT footer EMPTY>
  <!ATTLIST section class NMTOKEN #IMPLIED>
  <!ATTLIST footer created CDATA #IMPLIED>
  <!ENTITY copy "&#169;">
]>
<data>
  <section class="intro">Text &copy;</section>
  <section>
    <subsection>Nested</subsection>
  </section>
  <footer created="2024"/>
</data>
```
