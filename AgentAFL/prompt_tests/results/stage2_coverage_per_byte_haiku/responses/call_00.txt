```xml
<?xml version="1.0"?>
<!DOCTYPE doc [
  <!NOTATION png PUBLIC "image/png" "">
  <!ELEMENT doc (item+)>
  <!ELEMENT item (#PCDATA)>
  <!ATTLIST item id ID #REQUIRED>
]>
<doc>
  <item id="first">Content</item>
</doc>
```

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE catalog [
  <!ENTITY % conditional "IGNORE">
  <!ELEMENT catalog (product)*>
  <!ELEMENT product (name,price)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT price (#PCDATA)>
  <!ATTLIST product sku CDATA #IMPLIED>
]>
<catalog>
  <product sku="123">
    <name>Widget</name>
    <price>9.99</price>
  </product>
</catalog>
```

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE data [
  <!ELEMENT data (record*)>
  <!ELEMENT record (field+)>
  <!ELEMENT field (#PCDATA)>
  <!ATTLIST record type (A|B|C) #REQUIRED>
  <!ATTLIST field xmlns:custom CDATA #FIXED "http://custom.ns">
]>
<data>
  <record type="A">
    <field>value1</field>
    <field>value2</field>
  </record>
</data>
```
