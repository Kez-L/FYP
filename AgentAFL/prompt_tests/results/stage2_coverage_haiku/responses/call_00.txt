```xml
<?xml version="1.0"?>
<!DOCTYPE config [
<!ELEMENT config (section*)>
<!ELEMENT section (param*)>
<!ELEMENT param EMPTY>
<!ATTLIST section name CDATA #REQUIRED>
<!ATTLIST param key CDATA #REQUIRED value CDATA #IMPLIED>
]>
<config>
  <section name="database">
    <param key="host" value="localhost"/>
    <param key="port" value="5432"/>
  </section>
  <section name="cache">
    <param key="ttl" value="3600"/>
  </section>
</config>
```

```xml
<!DOCTYPE html [
<!ELEMENT html (head, body)>
<!ELEMENT head (title)>
<!ELEMENT body (p|div)*>
<!ELEMENT title (#PCDATA)>
<!ELEMENT p (#PCDATA)>
<!ELEMENT div (p)*>
<!ATTLIST html xmlns CDATA #FIXED "http://www.w3.org/1999/xhtml">
<!ATTLIST div class CDATA #IMPLIED id ID #IMPLIED>
]>
<html xmlns="http://www.w3.org/1999/xhtml">
  <head><title>Test Page</title></head>
  <body>
    <p>Paragraph text</p>
    <div class="container" id="main">
      <p>Nested paragraph</p>
    </div>
  </body>
</html>
```

```xml
<?xml version="1.0" encoding="ISO-8859-1"?>
<!DOCTYPE data [
<!ELEMENT data (record*)>
<!ELEMENT record (field+)>
<!ELEMENT field (#PCDATA)>
<!ATTLIST record id ID #REQUIRED type (A|B|C) "A">
<!ATTLIST field name CDATA #REQUIRED>
<!ENTITY lt "&#60;">
<!ENTITY gt "&#62;">
]>
<data>
  <record id="r1" type="A">
    <field name="first">Value &lt;1&gt;</field>
    <field name="second">Text with &gt; symbol</field>
  </record>
  <record id="r2" type="B">
    <field name="data">Another entry</field>
  </record>
</data>
```

```xml
<!DOCTYPE manifest [
<!ELEMENT manifest (application)>
<!ELEMENT application (activity*, service*)>
<!ELEMENT activity EMPTY>
<!ELEMENT service EMPTY>
<!ATTLIST manifest package CDATA #REQUIRED>
<!ATTLIST activity android:name CDATA #REQUIRED android:label CDATA #IMPLIED>
<!ATTLIST service android:name CDATA #REQUIRED android:enabled (true|false) "true">
]>
<manifest package="com.example.app">
  <application>
    <activity android:name=".MainActivity" android:label="Main"/>
    <activity android:name=".SettingsActivity"/>
    <service android:name=".MyService" android:enabled="true"/>
  </application>
</manifest>
```
