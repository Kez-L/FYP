<!DOCTYPE feed [

<!ELEMENT link EMPTY>

<!ELEMENT id (#PCDATA)>

<!ELEMENT span (#PCDATA)>
<!ELEMENT published (#PCDATA)>

<!ATTLIST link rel CDATA #REQUIRED href CDATA #REQUIRED type CDATA #IMPLIED>
<!ATTLIST entry updated CDATA #IMPLIED>
<!ATTLIST content type (text|html|xhtml) "text">
<!ATTLIST span class CDATA #IMPLIED>
]>

  <title>Example Feed</title>
  <link rel="self" href="http://example.org/feed" type="application/atom+xml"/>

    <id>urn:uuid:1234</id>
    <title>First Entry</title>
    <author>John</author>
    <content type="html">Hello <span class="highlight">World</span></content>
