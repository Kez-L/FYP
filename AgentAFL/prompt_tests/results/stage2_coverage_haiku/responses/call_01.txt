```xml
<!ELEMENT h2 (#PCDATA)>
<!ELEMENT p (#PCDATA|emphasis)*>
<!ELEMENT emphasis (#PCDATA)>

<!ENTITY fig1 SYSTEM "image.jpg" NDATA jpeg>
]>

  <title>Test Article</title>
  <body>
    <section>
      <h2>Introduction</h2>
      <p>This is a <emphasis>test</emphasis> paragraph.</p>
      <p>Another paragraph with content.</p>
    </section>
  </body>
</article>
```

```xml
<?xml version="1.0" encoding="ISO-8859-1"?>
<!DOCTYPE config [
<!ELEMENT config (database|cache|logging)*>
<!ELEMENT database (host, port, credentials)>
<!ELEMENT cache (ttl, maxsize)>
<!ELEMENT logging (level, output)>
<!ELEMENT host (#PCDATA)>
<!ELEMENT port (#PCDATA)>
<!ELEMENT credentials (#PCDATA)>
<!ELEMENT ttl (#PCDATA)>
<!ELEMENT maxsize (#PCDATA)>
<!ELEMENT level (#PCDATA)>
<!ELEMENT output (#PCDATA)>
<!ATTLIST database type (mysql|postgres) #REQUIRED>
<!ATTLIST cache enabled (yes|no) "yes">
<!ATTLIST logging format CDATA #IMPLIED>
]>
<config>
  <database type="postgres">
    <host>db.example.com</host>
    <port>5432</port>
    <credentials>user:pass</credentials>
  </database>
  <cache enabled="yes">
    <ttl>3600</ttl>
    <maxsize>1000</maxsize>
  </cache>
  <logging format="json">
    <level>INFO</level>
    <output>/var/log/app.log</output>
  </logging>
</config>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE manifest [
<!ELEMENT manifest (application, permissions?, features?)>
<!ELEMENT application (activity+)>
<!ELEMENT activity (intent-filter?)>
<!ELEMENT intent-filter (action, category*)>
<!ELEMENT action EMPTY>
<!ELEMENT category EMPTY>
<!ELEMENT permissions (permission*)>
<!ELEMENT permission EMPTY>
<!ELEMENT features (feature*)>
<!ELEMENT feature EMPTY>
<!ATTLIST application name CDATA #REQUIRED package CDATA #REQUIRED>
<!ATTLIST activity name CDATA #REQUIRED>
<!ATTLIST action name CDATA #REQUIRED>
<!ATTLIST category name CDATA #REQUIRED>
<!ATTLIST permission name CDATA #REQUIRED>
<!ATTLIST feature name CDATA #REQUIRED required (true|false) "false">
]>
<manifest package="com.example.app" name="TestApp">
  <application name="MainActivity" package="com.example.app">
    <activity name=".MainActivity">
      <intent-filter>
        <action name="android.intent.action.MAIN"/>
        <category name="android.intent.category.LAUNCHER"/>
      </intent-filter>
    </activity>
  </application>
  <permissions>
    <permission name="android.permission.INTERNET"/>
    <permission name="android.permission.ACCESS_NETWORK_STATE"/>
  </permissions>
  <features>
    <feature name="android.hardware.camera" required="false"/>
  </features>
</manifest>
```
