<!DOCTYPE document [

<!ELEMENT section (title, (paragraph|list)+)>
<!ELEMENT title (#PCDATA)>
<!ELEMENT paragraph (#PCDATA | emphasis)*>
<!ELEMENT emphasis (#PCDATA)>
<!ELEMENT list (item+)>
<!ELEMENT item (#PCDATA)>
<!ATTLIST document version CDATA #FIXED "1.0">
<!ATTLIST section level (1|2|3) "1">
<!ATTLIST list type (ordered|unordered) "unordered">
<!NOTATION svg PUBLIC "image/svg+xml">
]>
<document version="1.0">
  <section level="1">
    <title>Main Section</title>
    <paragraph>Text with <emphasis>emphasis</emphasis></paragraph>
    <list type="unordered">
      <item>First item</item>
      <item>Second item</item>
    </list>
  </section>
</document>

<!ELEMENT name (#PCDATA)>
<!ELEMENT price (#PCDATA)>
<!ELEMENT stock (#PCDATA)>

]>

  <product sku="PROD001" inStock="yes">
    <name>Widget</name>
    <price currency="USD">29.99</price>
    <stock>100</stock>
  </product>
  <product sku="PROD002" inStock="no">
    <name>Gadget</name>
    <price currency="EUR">49.50</price>

  </product>
</catalog>

<!ELEMENT id (#PCDATA)>
<!ELEMENT updated (#PCDATA)>
<!ELEMENT content (#PCDATA)>
<!ATTLIST feed xmlns CDATA #FIXED "http://www.w3.org/2005/Atom">

<!ATTLIST content type (text|html|xhtml) "text">
<!ENTITY copy "&#169;">
]>
<feed xmlns="http://www.w3.org/2005/Atom">
  <entry published="2024-01-01">
    <id>tag:example.com,2024:entry1</id>
    <updated>2024-01-01T12:00:00Z</updated>
    <content type="html">&lt;p&gt;Content&lt;/p&gt;</content>
  </entry>
</feed>
