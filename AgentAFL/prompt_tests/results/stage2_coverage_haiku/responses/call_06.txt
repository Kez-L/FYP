```xml
<!DOCTYPE config [
<!ELEMENT config (database | cache | logging)*>
<!ELEMENT database (connection+)>
<!ELEMENT cache (server+)>
<!ELEMENT logging (level, output)>
<!ELEMENT connection (host, port, credentials)>
<!ELEMENT server (#PCDATA)>
<!ELEMENT host (#PCDATA)>
<!ELEMENT port (#PCDATA)>
<!ELEMENT credentials (#PCDATA)>
<!ELEMENT level (#PCDATA)>
<!ELEMENT output (#PCDATA)>
<!ATTLIST database type CDATA #REQUIRED version CDATA #IMPLIED>
<!ATTLIST cache ttl CDATA "3600">
<!ATTLIST connection timeout CDATA #IMPLIED pool_size NMTOKEN #IMPLIED>
<!NOTATION jpeg PUBLIC "image/jpeg">
]>
<config>
  <database type="postgresql" version="13">
    <connection timeout="30" pool_size="10">
      <host>db.internal</host>
      <port>5432</port>
      <credentials>user:pass</credentials>
    </connection>
  </database>
  <cache ttl="7200">
    <server>cache1.local</server>
    <server>cache2.local</server>
  </cache>
  <logging>
    <level>DEBUG</level>
    <output>/var/log/app.log</output>
  </logging>
</config>
```

```xml
<?xml version="1.0" standalone="no"?>
<!DOCTYPE feed [
<!ELEMENT feed (entry*)>
<!ELEMENT entry (id, title, link, published, author*)>
<!ELEMENT id (#PCDATA)>
<!ELEMENT title (#PCDATA)>
<!ELEMENT link EMPTY>
<!ELEMENT published (#PCDATA)>
<!ELEMENT author (name, email?)>
<!ELEMENT name (#PCDATA)>
<!ELEMENT email (#PCDATA)>
<!ATTLIST feed version CDATA #FIXED "1.0" xmlns CDATA #IMPLIED>
<!ATTLIST link href CDATA #REQUIRED rel CDATA #IMPLIED type CDATA #IMPLIED>
<!ATTLIST entry updated CDATA #IMPLIED status (published|draft|archived) "published">
<!ENTITY nbsp "&#160;">
<!ENTITY copy "&#169;">
]>
<feed version="1.0" xmlns="http://www.w3.org/2005/Atom">
  <entry updated="2024-01-15" status="published">
    <id>urn:uuid:1234-5678-90ab</id>
    <title>First Entry Title</title>

    <published>2024-01-15T09:00:00Z</published>

  </entry>
```
