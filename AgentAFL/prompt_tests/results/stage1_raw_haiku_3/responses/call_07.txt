```xml
<?xml version="1.0"?>
<!DOCTYPE root [
  <!ELEMENT root (child*)>
  <!ELEMENT child (#PCDATA)>
  <!ATTLIST child id ID #REQUIRED>
  <!ENTITY internal "test">
]>
<root>
  <child id="c1">&internal;</child>
  <child id="c2">text</child>
</root>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE doc [
  <!ENTITY % param1 "value1">
  <!ENTITY % param2 "%param1;">
  <!ELEMENT doc (section+)>
  <!ELEMENT section (title, content)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT content (#PCDATA | emphasis)*>
  <!ELEMENT emphasis (#PCDATA)>
]>
<doc>
  <section>
    <title>First</title>
    <content>Some content with <emphasis>emphasis</emphasis></content>
  </section>
</doc>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE messages [
  <!ELEMENT messages (message*)>
  <!ELEMENT message (from, to, body)>
  <!ELEMENT from (#PCDATA)>
  <!ELEMENT to (#PCDATA)>
  <!ELEMENT body (#PCDATA)>
  <!ATTLIST message id CDATA #IMPLIED>
  <!ENTITY sender "alice">
]>
<messages>
  <message id="1">
    <from>&sender;</from>
    <to>bob</to>
    <body>Hello</body>
  </message>
</messages>
```
