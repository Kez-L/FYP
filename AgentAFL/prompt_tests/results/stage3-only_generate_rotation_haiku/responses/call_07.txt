```xml
<?xml version="1.0"?>
<!DOCTYPE root [
  <!ELEMENT root (child+)>
  <!ELEMENT child (subchild*)>
  <!ELEMENT subchild (#PCDATA)>
  <!ATTLIST root xmlns CDATA #IMPLIED>
  <!ENTITY nbsp "&#160;">
  <!ENTITY copy "&#169;">
  <!ENTITY % entities "<!ENTITY &#x25; recursive SYSTEM 'urn:recursive'>">
  %entities;
]>
<root xmlns="http://example.com/ns">
  <child id="1">&nbsp;&copy;</child>
  <child id="2">Text with&nbsp;entity</child>
</root>
```

```xml
  <!ENTITY % end "]]>">

    <item>&textblock;</item>
    <item>Normal text</item>
  </section>
```
