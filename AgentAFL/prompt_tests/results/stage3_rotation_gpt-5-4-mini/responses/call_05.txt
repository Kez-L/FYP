</somedoc>

```xml
<?xml version="1.0"?>
<!DOCTYPE root [
  <!ELEMENT item (#PCDATA)>
  <!ELEMENT root (item+)>
  <!ENTITY reader "&writer; parser">
  <!ENTITY writer "xml">
]>
<root>
  <item>one</item>
  <item>two &amp; three</item>
  <item>&reader;</item>
</root>
```
