```xml
<?xml version="1.0"?>
<!DOCTYPE catalog [
  <!ELEMENT catalog (entry+)>
  <!ELEMENT entry (#PCDATA|subentry)*>
  <!ELEMENT subentry EMPTY>
  <!ATTLIST entry name CDATA #REQUIRED>
  <!ENTITY a "AAA">
  <!ENTITY b "&a;&a;">
]>
<catalog>
  <entry name="alpha">&b;<subentry/></entry>
  <entry name="beta">text&#x0A;&#x09;&a;</entry>
</catalog>
```
