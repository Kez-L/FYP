```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE doc [
  <!ELEMENT doc (item+)>
  <!ELEMENT item (#PCDATA)>
  <!ATTLIST item type CDATA #IMPLIED>
  <!NOTATION png SYSTEM "image/png">
]>
<doc>
  <item type="text">First item</item>
  <item type="data">Second item</item>
</doc>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE config [
  <!ELEMENT config (section*)>
  <!ELEMENT section (param*)>
  <!ELEMENT param EMPTY>
  <!ATTLIST param name NMTOKEN #REQUIRED>
  <!ATTLIST param value CDATA #REQUIRED>
]>
<config>
  <section>
    <param name="timeout" value="30"/>
    <param name="retries" value="3"/>
  </section>
</config>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE html [
  <!ELEMENT html (head, body)>
  <!ELEMENT head (title)>
  <!ELEMENT title (#PCDATA)>
  <!ELEMENT body (p|div)*>
  <!ELEMENT p (#PCDATA|span)*>
  <!ELEMENT div (#PCDATA)>
  <!ELEMENT span (#PCDATA)>
]>
<html>
  <head><title>Test Page</title></head>
  <body>
    <p>Hello <span>world</span></p>
    <div>Content here</div>
  </body>
</html>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE manifest [
  <!ELEMENT manifest (package+)>
  <!ELEMENT package (version, author?)>
  <!ELEMENT version (#PCDATA)>
  <!ELEMENT author (#PCDATA)>
  <!ATTLIST package name ID #REQUIRED>
  <!ATTLIST package xmlns CDATA #FIXED "http://example.com">
]>
<manifest xmlns="http://example.com">
  <package name="pkg1">
    <version>1.0.0</version>
    <author>John Doe</author>
  </package>
  <package name="pkg2">
    <version>2.1.0</version>
  </package>
</manifest>
```
