```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE config [
  <!ELEMENT config (setting)*>
  <!ELEMENT setting (#PCDATA)>
  <!ATTLIST setting name CDATA #REQUIRED>
  <!NOTATION png PUBLIC "image/png">
  <!ENTITY % param "value">
]>
<config>
  <setting name="key1">data1</setting>
  <setting name="key2">data2</setting>
</config>
```

```xml
<!DOCTYPE html [
  <!ELEMENT html (head, body)>
  <!ELEMENT head (title)>
  <!ELEMENT body (p|div)*>
  <!ELEMENT p (#PCDATA)>
  <!ELEMENT div (#PCDATA|p)*>
  <!ELEMENT title (#PCDATA)>
]>
<html>
  <head><title>Test</title></head>
  <body>
    <div><p>Content</p></div>
  </body>
</html>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE data [
  <!ELEMENT data (record+)>
  <!ELEMENT record (field*)>
  <!ELEMENT field (#PCDATA)>
  <!ATTLIST record id CDATA #IMPLIED type CDATA #IMPLIED>
  <!ATTLIST field name CDATA #REQUIRED>
]>
<data>
  <record id="1" type="A">
    <field name="f1">value1</field>
    <field name="f2">value2</field>
  </record>
</data>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE manifest [
  <!ELEMENT manifest (package+)>
  <!ELEMENT package (component*)>
  <!ELEMENT component EMPTY>
  <!ATTLIST package name CDATA #REQUIRED version CDATA #REQUIRED>
  <!ATTLIST component class CDATA #REQUIRED enabled (true|false) "true">
]>
<manifest>
  <package name="app" version="1.0">
    <component class="MainActivity" enabled="true"/>
    <component class="Service" enabled="false"/>
  </package>
</manifest>
```
