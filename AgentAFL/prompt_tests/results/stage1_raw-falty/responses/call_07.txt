```xml
<?xml version="1.0"?>
<!DOCTYPE root [
  <!ELEMENT root (child*)>
  <!ELEMENT child (#PCDATA)>
  <!ATTLIST child id ID #REQUIRED>
]>
<root>
  <child id="c1">text content</child>
  <child id="c2">more text</child>
</root>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE test [
  <!ELEMENT test (a|b|c)+>
  <!ELEMENT a (#PCDATA)>
  <!ELEMENT b (a,c)>
  <!ELEMENT c EMPTY>
  <!ATTLIST c value CDATA #IMPLIED>
]>
<test>
  <a>first</a>
  <b><a>nested</a><c value="empty"/></b>
  <c value="standalone"/>
</test>
```
