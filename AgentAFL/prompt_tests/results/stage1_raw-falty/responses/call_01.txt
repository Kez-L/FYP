  <!ATTLIST setting name CDATA #REQUIRED>
  <!ATTLIST setting type (string|int|bool) "string">
]>

  <setting name="timeout" type="int">3000</setting>
  <setting name="debug" type="bool">true</setting>
  <setting name="path" type="string">/usr/local/bin</setting>
  <setting name="empty"/>
</config>
