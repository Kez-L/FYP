```xml
<?xml version="1.0"?>
<!DOCTYPE note [
  <!ELEMENT note (to,from,body)>
  <!ELEMENT to (#PCDATA)>
  <!ELEMENT from (#PCDATA)>
  <!ELEMENT body (#PCDATA)>
]>
<note>
  <to>Alice</to>
  <from>Bob</from>
  <body>Hello world</body>
</note>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE catalog [
  <!ELEMENT catalog (item+)>
  <!ELEMENT item (name,price,desc?)>
  <!ELEMENT name (#PCDATA)>
  <!ELEMENT price (#PCDATA)>
  <!ELEMENT desc (#PCDATA)>
]>
<catalog>
  <item id="1">
    <name>Widget</name>
    <price currency="USD">19.99</price>
    <desc>Standard widget</desc>
  </item>
  <item id="2">
    <name>Gadget</name>
    <price currency="EUR">29.50</price>
  </item>
</catalog>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE data [
  <!ATTLIST record kind CDATA #REQUIRED>
  <!ATTLIST record flag (yes|no) "no">
]>
<data>
  <record kind="alpha" flag="yes">
    <value>123</value>
  </record>
  <record kind="beta">
    <value>456</value>
  </record>
</data>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE library [
  <!ELEMENT library (section*)>
  <!ELEMENT section (book*)>
  <!ELEMENT book EMPTY>
  <!ATTLIST book title CDATA #IMPLIED>
  <!ATTLIST book author CDATA #IMPLIED>
]>
<library>
  <section name="fiction">
    <book title="Dune" author="Frank Herbert"/>
    <book title="Neuromancer" author="William Gibson"/>
  </section>
  <section name="reference"/>
</library>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE root [
  <!ENTITY greeting "hi">
  <!ELEMENT root (msg|empty)*>
  <!ELEMENT msg (#PCDATA)>
  <!ELEMENT empty EMPTY>
]>
<root>
  <msg>&greeting;, parser!</msg>
  <empty/>
  <msg xml:lang="en">Nested <b>text</b> not allowed in DTD, but present</msg>
</root>
```
