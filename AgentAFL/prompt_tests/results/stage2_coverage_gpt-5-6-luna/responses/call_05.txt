<!DOCTYPE query [

  <!ELEMENT query (select, source+, constraint*)>
  <!ELEMENT select (field+)>
  <!ELEMENT field EMPTY>
  <!ATTLIST field
    name %common; #REQUIRED

    aggregate (none|count|min|max|sum) "none">
  <!ELEMENT source EMPTY>
  <!ATTLIST source
    table %common; #REQUIRED
    alias %common; #IMPLIED>
  <!ELEMENT constraint (left, operator, right)>
  <!ELEMENT left (#PCDATA)>
  <!ELEMENT operator (#PCDATA)>
  <!ELEMENT right (#PCDATA)>
  <!ENTITY quote "&quot;">
  <!ENTITY wildcard "*">
]>
<query xmlns:q="urn:query" q:version="3">

    <field name="id" aggregate="count"/>

  </select>

    <left>r.status</left>

  </constraint>
