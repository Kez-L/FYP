```xml
<!DOCTYPE schema [
<!ELEMENT schema (element|attribute|complexType)*>
<!ELEMENT element (annotation?, restriction?)>
<!ELEMENT attribute (annotation?)>
<!ELEMENT complexType (sequence?, annotation?)>
<!ELEMENT sequence (element+)>
<!ELEMENT annotation (documentation)>
<!ELEMENT documentation (#PCDATA)>
<!ELEMENT restriction (enumeration*)>
<!ELEMENT enumeration EMPTY>
<!ATTLIST element name CDATA #REQUIRED type CDATA #IMPLIED minOccurs CDATA #IMPLIED maxOccurs CDATA #IMPLIED>
<!ATTLIST attribute name CDATA #REQUIRED type CDATA #REQUIRED use (required|optional) "optional">
<!ATTLIST complexType name CDATA #REQUIRED>
<!ATTLIST enumeration value CDATA #REQUIRED>
<!ATTLIST restriction base CDATA #REQUIRED>
]>
<schema>
  <complexType name="PersonType">
    <sequence>
      <element name="firstname" type="string" minOccurs="1"/>
      <element name="lastname" type="string" minOccurs="1"/>
      <element name="age" type="integer" minOccurs="0" maxOccurs="1"/>
    </sequence>
  </complexType>
  <element name="status" type="string">
    <restriction base="string">
      <enumeration value="active"/>
      <enumeration value="inactive"/>
      <enumeration value="pending"/>
    </restriction>
  </element>
  <attribute name="id" type="ID" use="required">
    <annotation>
      <documentation>Unique identifier</documentation>
    </annotation>
  </attribute>
</schema>
```

```xml
<?xml version="1.0" encoding="ISO-8859-1"?>
<!DOCTYPE memo [
<!ELEMENT memo (header, body, footer?)>
<!ELEMENT header (to, from, date, subject)>
<!ELEMENT body (paragraph|section)+>
<!ELEMENT paragraph (#PCDATA|emphasis|strong)*>
<!ELEMENT section (title, paragraph+)>
<!ELEMENT title (#PCDATA)>
<!ELEMENT emphasis (#PCDATA)>
<!ELEMENT strong (#PCDATA)>
<!ELEMENT to (#PCDATA)>
<!ELEMENT from (#PCDATA)>
<!ELEMENT date (#PCDATA)>
<!ELEMENT subject (#PCDATA)>
<!ELEMENT footer (signature, stamp?)>
<!ELEMENT signature (#PCDATA)>
<!ELEMENT stamp EMPTY>
<!ATTLIST memo priority (high|medium|low) "medium" classification (public|confidential|restricted) #REQUIRED>
<!ATTLIST stamp date CDATA #REQUIRED>
]>
<memo priority="high" classification="confidential">
  <header>
    <to>Team Lead</to>
    <from>Assistant</from>
    <date>2024-01-15</date>
    <subject>Quarterly Review</subject>
  </header>
  <body>
    <paragraph>This is an important update regarding our <emphasis>project status</emphasis>.</paragraph>
    <section>
      <title>Accomplishments</title>

    </section>
    <section>
      <title>Next Steps</title>

    </section>
```
