```xml
<!ATTLIST item id ID #REQUIRED xmlns:custom CDATA #IMPLIED>
<!ATTLIST child ref IDREF #REQUIRED>
<!ENTITY testent "expansion">
]>
<root>
  <item id="i1" xmlns:custom="http://test.org">

  </item>

</root>
```

```xml
<!DOCTYPE article [
<!ELEMENT article (title, body, footnote*)>
<!ELEMENT title (#PCDATA)>
<!ELEMENT body (paragraph+)>
<!ELEMENT paragraph (#PCDATA|emphasis|strong)*>
<!ELEMENT emphasis (#PCDATA)>
<!ELEMENT strong (#PCDATA)>
<!ELEMENT footnote (#PCDATA)>
<!ATTLIST article version CDATA #FIXED "2.0" status (draft|published|archived) "draft">
<!ATTLIST paragraph xml:space (default|preserve) "default">
]>
<article version="2.0" status="published">
  <title>XML Parser Edge Cases</title>
  <body>
    <paragraph xml:space="preserve">
      This has   <emphasis>emphasis</emphasis> and <strong>strong</strong> text.
    </paragraph>
    <paragraph>Another paragraph with special chars: &lt; &gt; &amp; &quot;</paragraph>
  </body>
  <footnote>First note</footnote>
  <footnote>Second note</footnote>
</article>
```

```xml
<?xml version="1.0" encoding="UTF-16"?>
<!DOCTYPE catalog [
<!ELEMENT catalog (item+)>
<!ELEMENT item (price, quantity)>
<!ELEMENT price (#PCDATA)>
<!ELEMENT quantity (#PCDATA)>
<!ATTLIST item sku CDATA #REQUIRED currency (USD|EUR|GBP) #IMPLIED in_stock (yes|no) #DEFAULT "yes">
<!NOTATION image PUBLIC "image/png">
<!ENTITY logo SYSTEM "logo.png" NDATA image>
]>
<catalog>
  <item sku="PROD-001" currency="USD" in_stock="yes">
    <price>19.99</price>
    <quantity>5</quantity>
  </item>
  <item sku="PROD-002" currency="EUR">
    <price>29.50</price>
    <quantity>0</quantity>
  </item>
</catalog>
```

```xml
<!DOCTYPE schema [
<!ELEMENT schema (element|attribute)*>
<!ELEMENT element (restriction?)>
<!ELEMENT attribute (enumeration*)>
<!ELEMENT restriction (#PCDATA)>
<!ELEMENT enumeration EMPTY>
<!ATTLIST element name NMTOKEN #REQUIRED type NMTOKEN "string">
<!ATTLIST attribute name NMTOKEN #REQUIRED use (required|optional|prohibited) "optional">
<!ATTLIST enumeration value CDATA #REQUIRED default CDATA #IMPLIED>
]>
<schema>
  <element name="user" type="complex">
    <restriction>maxLength=100</restriction>
  </element>
  <attribute name="id" use="required">
    <enumeration value="v1" default="true"/>
    <enumeration value="v2"/>
  </attribute>
  <element name="timestamp" type="dateTime"/>
</schema>
```

```xml
<?xml version="1.0" standalone="yes"?>
<!DOCTYPE data [
<!ELEMENT data (record*)>
<!ELEMENT record (field+)>
<!ELEMENT field (#PCDATA)>
<!ATTLIST record id ID #REQUIRED timestamp CDATA #FIXED "2024-01-01T00:00:00Z" xmlns:meta CDATA #IMPLIED>
<!ATTLIST field name CDATA #REQUIRED type (text|number|boolean) "text" optional (true|false) "false">
<!ENTITY % fieldContent "(field)*">
]>
<data>
  <record id="rec001" timestamp="2024-01-01T00:00:00Z">
    <field name="username" type="text">alice</field>
    <field name="age" type="number" optional="false">42</field>
    <field name="active" type="boolean">true</field>
  </record>
  <record id="rec002" timestamp="2024-01-01T00:00:00Z">
    <field name="username" type="text">bob</field>
    <field name="age" type="number">38</field>
  </record>
</data>
```
