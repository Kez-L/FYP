```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE catalog [
<!ELEMENT catalog (product+)>
<!ELEMENT product (name, price, inventory)>
<!ELEMENT name (#PCDATA)>
<!ELEMENT price (#PCDATA)>
<!ELEMENT inventory EMPTY>
<!ATTLIST product id ID #REQUIRED sku CDATA #IMPLIED>
<!ATTLIST inventory count CDATA #REQUIRED location (warehouse|store|transit) "warehouse">
]>
<catalog>
  <product id="p001" sku="SKU-12345">
    <name>Widget A</name>
    <price currency="USD">29.99</price>
    <inventory count="150" location="warehouse"/>
  </product>
  <product id="p002">
    <name>Gadget B</name>
    <price>49.50</price>
    <inventory count="0" location="store"/>
  </product>
</catalog>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE network [
<!ELEMENT network (device+)>
<!ELEMENT device (interface+, routes?)>
<!ELEMENT interface (ipv4?, ipv6?)>
<!ELEMENT ipv4 EMPTY>
<!ELEMENT ipv6 EMPTY>
<!ELEMENT routes (route*)>
<!ELEMENT route EMPTY>
<!ATTLIST device hostname CDATA #REQUIRED type (router|switch|host) "host">
<!ATTLIST interface name CDATA #REQUIRED mtu CDATA #DEFAULT "1500">
<!ATTLIST ipv4 address CDATA #REQUIRED netmask CDATA #REQUIRED>
<!ATTLIST ipv6 address CDATA #REQUIRED prefix CDATA #REQUIRED>
<!ATTLIST route destination CDATA #REQUIRED gateway CDATA #REQUIRED metric CDATA #IMPLIED>
]>
<network>
  <device hostname="router01" type="router">
    <interface name="eth0" mtu="1500">
      <ipv4 address="192.168.1.1" netmask="255.255.255.0"/>
      <ipv6 address="2001:db8::1" prefix="64"/>
    </interface>
    <routes>
      <route destination="10.0.0.0/8" gateway="192.168.1.254" metric="1"/>
    </routes>
  </device>
  <device hostname="host01" type="host">
    <interface name="eth0" mtu="1500">
      <ipv4 address="192.168.1.100" netmask="255.255.255.0"/>
    </interface>
  </device>
</network>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE vocabulary [
<!ELEMENT vocabulary (word+)>
<!ELEMENT word (definition+, examples, etymology?)>
<!ELEMENT definition (#PCDATA)>
<!ELEMENT examples (example+)>
<!ELEMENT example (#PCDATA)>
<!ELEMENT etymology (#PCDATA)>
<!ATTLIST vocabulary lang CDATA #REQUIRED version CDATA #IMPLIED>
<!ATTLIST word partOfSpeech (noun|verb|adjective|adverb|preposition) #REQUIRED frequency CDATA #IMPLIED>
<!ATTLIST definition order CDATA #REQUIRED context CDATA #IMPLIED>
<!ATTLIST example usage CDATA #IMPLIED>
]>
<vocabulary lang="en" version="1.0">
  <word partOfSpeech="noun" frequency="high">
    <definition order="1" context="general">A small domesticated carnivorous mammal.</definition>
    <definition order="2" context="informal">A despicable person.</definition>
    <examples>
      <example usage="literal">The cat sat on the mat.</example>
      <example usage="figurative">Don't be such a cat about it.</example>
    </examples>
    <etymology>From Old English catt, from Latin cattus.</etymology>
  </word>
  <word partOfSpeech="verb" frequency="medium">
    <definition order="1">To consume food.</definition>
    <examples>
      <example>We eat breakfast at 7 AM.</example>
    </examples>
  </word>
</vocabulary>
```
