```xml
<!ELEMENT title (#PCDATA)>
<!ELEMENT paragraph (#PCDATA|emphasis|strong)*>
<!ELEMENT emphasis (#PCDATA)>
<!ELEMENT strong (#PCDATA)>

]>
<doc>
  <section level="1" class="intro">
    <title>Introduction</title>
    <paragraph>This is <emphasis>important</emphasis> and <strong>bold</strong>.</paragraph>
  </section>
```

```xml
<!ELEMENT title (#PCDATA)>
<!ELEMENT isbn (#PCDATA)>

]>
<library>
  <book published="2020-01-15" year="2020" category="fiction">
```

```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE config [
<!ELEMENT config (setting+)>
<!ELEMENT setting (name, value, description?)>
<!ELEMENT name (#PCDATA)>
<!ELEMENT value (#PCDATA)>
<!ELEMENT description (#PCDATA)>
<!ATTLIST setting type (string|integer|boolean) "string" required (yes|no) "no">
]>
<config>
  <setting type="string" required="yes">
    <name>appname</name>
    <value>MyApplication</value>
    <description>The application name</description>
  </setting>
  <setting type="integer">
    <name>port</name>
    <value>8080</value>
  </setting>
  <setting type="boolean" required="no">
    <name>debug</name>
    <value>false</value>
  </setting>
</config>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE manifest [
<!ELEMENT manifest (metadata, packages)>
<!ELEMENT metadata (name, version, author*)>
<!ELEMENT name (#PCDATA)>
<!ELEMENT version (#PCDATA)>
<!ELEMENT author (#PCDATA)>
<!ELEMENT packages (package+)>
<!ELEMENT package (dependency*)>
<!ELEMENT dependency (#PCDATA)>
<!ATTLIST metadata timestamp CDATA #IMPLIED>
<!ATTLIST package id ID #REQUIRED status (active|inactive) "active">
<!ATTLIST dependency version NMTOKEN #IMPLIED type (required|optional) "required">
]>
<manifest>
  <metadata timestamp="2024-01-15T10:30:00Z">
    <name>ProjectName</name>
    <version>1.2.3</version>
    <author>Developer One</author>
    <author>Developer Two</author>
  </metadata>
  <packages>
    <package id="pkg1" status="active">
      <dependency version="2.0" type="required">libxml2</dependency>
      <dependency type="optional">openssl</dependency>
    </package>
    <package id="pkg2" status="inactive">
      <dependency version="1.5">zlib</dependency>
    </package>
  </packages>
</manifest>
```
