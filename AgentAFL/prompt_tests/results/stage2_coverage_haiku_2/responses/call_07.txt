```xml
<!ELEMENT param EMPTY>
<!ATTLIST section name CDATA #REQUIRED enabled (yes|no) "yes">
<!ATTLIST param key ID #REQUIRED value CDATA #REQUIRED default CDATA #IMPLIED>
<!ELEMENT metadata (created, modified)>
<!ELEMENT created (#PCDATA)>
<!ELEMENT modified (#PCDATA)>
]>
<config>
  <section name="database" enabled="yes">
    <param key="host" value="localhost" default="127.0.0.1"/>
    <param key="port" value="5432"/>
    <param key="user" value="admin"/>
  </section>
  <section name="logging" enabled="no">
    <param key="level" value="INFO"/>
    <param key="file" value="/var/log/app.log"/>
  </section>
</config>
```

```xml
<!DOCTYPE message [
<!ELEMENT message (header, payload, signature?)>
<!ELEMENT header (from, to, timestamp)>
<!ELEMENT from (#PCDATA)>
<!ELEMENT to (#PCDATA)>
<!ELEMENT timestamp (#PCDATA)>
<!ELEMENT payload (text|binary)*>
<!ELEMENT text (#PCDATA)>
<!ELEMENT binary EMPTY>
<!ELEMENT signature (#PCDATA)>
<!ATTLIST message version CDATA #FIXED "1.0" id CDATA #REQUIRED>
<!ATTLIST header priority (high|medium|low) "medium">
<!ATTLIST binary encoding (base64|hex) "base64" size CDATA #REQUIRED>
]>
<message version="1.0" id="msg-12345">
  <header priority="high">
    <from>sender@example.com</from>
    <to>recipient@example.com</to>
    <timestamp>2024-01-15T10:30:00Z</timestamp>
  </header>
  <payload>
    <text>Important message content</text>
    <binary encoding="base64" size="1024"/>
  </payload>
  <signature>ABCD1234EFGH5678</signature>
</message>
```

```xml
<?xml version="1.0"?>
<!DOCTYPE catalog [
<!ELEMENT catalog (product+, category+)>
<!ELEMENT product (name, description, price, stock)>
<!ELEMENT name (#PCDATA)>
<!ELEMENT description (#PCDATA)>
<!ELEMENT price (#PCDATA)>
<!ELEMENT stock (#PCDATA)>
<!ELEMENT category (label, subcategory*)>
<!ELEMENT label (#PCDATA)>
<!ELEMENT subcategory (#PCDATA)>
<!ATTLIST product sku ID #REQUIRED category IDREF #REQUIRED available (true|false) "true">
<!ATTLIST category catid ID #REQUIRED parent IDREF #IMPLIED>
<!ATTLIST price currency CDATA #REQUIRED>
]>
<catalog>
  <product sku="PROD001" category="ELEC" available="true">
    <name>Laptop</name>
    <description>High performance laptop</description>
    <price currency="USD">999.99</price>
    <stock>15</stock>
  </product>
  <product sku="PROD002" category="ELEC" available="false">
    <name>Mouse</name>
    <description>Wireless mouse</description>

    <stock>0</stock>
  </product>
  <category catid="ELEC">
    <label>Electronics</label>

</catalog>
```
