```xml
<!DOCTYPE message [
<!ELEMENT message (header, body, footer?)>
<!ELEMENT header (from, to, subject)>
<!ELEMENT from (#PCDATA)>
<!ELEMENT to (#PCDATA)>
<!ELEMENT subject (#PCDATA)>
<!ELEMENT body (paragraph+)>
<!ELEMENT paragraph (#PCDATA|emphasis|strong)*>
<!ELEMENT emphasis (#PCDATA)>
<!ELEMENT strong (#PCDATA)>
<!ELEMENT footer (signature?)>
<!ELEMENT signature (#PCDATA)>
<!ATTLIST message id CDATA #REQUIRED priority (high|normal|low) "normal" xmlns:custom CDATA #IMPLIED>
<!ATTLIST paragraph indent CDATA #IMPLIED>
]>
<message id="msg-001" priority="high">
  <header>
    <from>sender@example.com</from>
    <to>recipient@example.com</to>
    <subject>Important Update</subject>
  </header>
  <body>
    <paragraph>This is a <emphasis>critical</emphasis> message.</paragraph>
    <paragraph indent="1">Please <strong>review</strong> immediately.</paragraph>
  </body>
  <footer>
    <signature>System Administrator</signature>
  </footer>
</message>
```

```xml
<?xml version="1.0" encoding="ISO-8859-1"?>

<!ELEMENT description (#PCDATA)>

<!ATTLIST price currency (USD|EUR|GBP) "USD" taxincluded (yes|no) "no">
```
